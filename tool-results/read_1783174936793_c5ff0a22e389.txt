   267→   267→function SummaryPanel({ selection, pricing, calc }: {
   268→   268→  selection: QuoteBuilderSelection
   269→   269→  pricing: PricingData
   270→   270→  calc: ReturnType<typeof calculateQuote> | null
   271→   271→}) {

   273→   273→  const selectedFrameworks = pricing.frameworks.filter((f) => selection.selectedFrameworkIds.includes(f.id))
   274→   274→  const tier = pricing.tiers.find((t) => t.id === selection.tierId)

   276→   276→  return (
   277→   277→    <div className="space-y-3">
   278→   278→      {calc.lines.filter((l) => l.lineType !== 'internal_time').length === 0 && (
   279→   279→        <p className="text-xs text-slate-400 py-4 text-center">
   280→   280→          Select framework(s) and a tier to see pricing

   282→   282→      )}

   285→   285→          {selectedFrameworks.length} framework{selectedFrameworks.length > 1 ? 's' : ''} selected
   286→   286→        </p>
   287→   287→      )}
   288→   288→
   289→   289→      {calc.lines.map((line, i) => {
   290→   290→        const isInternal = line.lineType === 'internal_time'
   291→   291→        const isDiscount = line.lineType === 'discount'
   292→   292→        return (
   293→   293→          <div
   294→   294→            key={i}
   295→   295→            className={cn(
   296→   296→              'flex items-start justify-between gap-2 text-sm',
   297→   297→              isInternal && 'opacity-60',

   301→   301→              {line.description}
   302→   302→            </span>
   303→   303→            <span className={cn(
   304→   304→              'font-medium tabular-nums shrink-0',
   305→   305→              isInternal ? 'text-slate-400 text-xs' : isDiscount ? 'text-rose-600' : 'text-slate-900',
   306→   306→            )}>
   307→   307→              {line.amountInr < 0 ? '-' : ''}{formatINR(Math.abs(line.amountInr))}
   308→   308→            </span>
   309→   309→          </div>
   310→   310→        )
   311→   311→      })}
   312→   312→
   313→   313→      <div className="border-t border-slate-200 pt-3 space-y-1.5">
   314→   314→        <div className="flex justify-between text-sm">
   315→   315→          <span className="text-slate-600">Subtotal</span>
   316→   316→          <span className="font-medium text-slate-900">{calc.billingCurrency === 'USD' ? formatUSD(calc.subtotalInr / calc.usdInrRate) : formatINR(calc.subtotalInr)}</span>
   317→   317→        </div>
   318→   318→        {calc.discountInr > 0 && (
   319→   319→          <div className="flex justify-between text-sm">
   320→   320→            <span className="text-rose-600">Discount ({calc.discountPct}%)</span>
   321→   321→            <span className="font-medium text-rose-600">-{calc.billingCurrency === 'USD' ? formatUSD(calc.discountInr / calc.usdInrRate) : formatINR(calc.discountInr)}</span>
   322→   322→          </div>
   323→   323→        )}
   324→   324→        {calc.gstAmountInr > 0 && (
   325→   325→          <div className="flex justify-between text-sm">
   326→   326→            <span className="text-slate-600">GST @ {calc.gstRate}%</span>

   357→   357→function FxRateCard({ currentRate }: { currentRate: number }) {
   358→   358→  const queryClient = useQueryClient()
   359→   359→  const [editing, setEditing] = useState(false)
   360→   360→  const [draft, setDraft] = useState(String(currentRate))
   361→   361→
   362→   362→  const updateFx = useMutation({
   363→   363→    mutationFn: (rate: number) => api.put('/admin/fx-rate', { rate }),
   364→   364→    onSuccess: () => {
   365→   365→      queryClient.invalidateQueries({ queryKey: ['pricing'] })
   366→   366→      toast.success('FX rate updated')
   367→   367→      setEditing(false)
   368→   368→    },
   369→   369→    onError: (e: Error) => toast.error(e.message),
   370→   370→  })
   371→   371→
   372→   372→  const handleSave = () => {
   373→   373→    const val = parseFloat(draft)
   374→   374→    if (isNaN(val) || val <= 0) {
   375→   375→      toast.error('Enter a valid positive rate')
   376→   376→      return
   377→   377→    }
   378→   378→    updateFx.mutate(val)
   379→   379→  }
   380→   380→
   381→   381→  const handleCancel = () => {
   382→   382→    setDraft(String(currentRate))
   383→   383→    setEditing(false)
   384→   384→  }
   385→   385→
   386→   386→  return (
   387→   387→    <div className="rounded-xl border border-slate-200 bg-white p-5">
   388→   388→      <div className="flex items-start justify-between gap-4">
   389→   389→        <div className="space-y-1">
   390→   390→          <h3 className="text-sm font-semibold text-slate-900">USD/INR Exchange Rate</h3>
   391→   391→          {!editing && (
   392→   392→            <p className="text-xs text-slate-500">This rate will be used to calculate USD equivalents</p>
   393→   393→          )}
   394→   394→        </div>
   395→   395→        {!editing ? (
   396→   396→          <div className="flex items-center gap-3">
   397→   397→            <span className="text-2xl font-bold text-brand-700 tabular-nums">{currentRate.toFixed(2)}</span>
   398→   398→            <button
   399→   399→              onClick={() => { setDraft(String(currentRate)); setEditing(true) }}
   400→   400→              className="inline-flex items-center gap-1.5 rounded-lg border border-brand-200 bg-brand-50 px-3 py-1.5 text-xs font-medium text-brand-700 hover:bg-brand-100 transition-colors"
   401→   401→            >
   402→   402→              <Pencil className="h-3 w-3" /> Update Rate
   403→   403→            </button>
   404→   404→          </div>
   405→   405→        ) : (
   406→   406→          <div className="flex items-center gap-2">
   407→   407→            <input
   408→   408→              type="number"
   409→   409→              step={0.01}
   410→   410→              min={0.01}
   411→   411→              value={draft}
   412→   412→              onChange={(e) => setDraft(e.target.value)}
   413→   413→              onKeyDown={(e) => { if (e.key === 'Enter') handleSave(); if (e.key === 'Escape') handleCancel() }}
   414→   414→              className="w-28 rounded-lg border border-slate-200 px-2.5 py-1.5 text-sm tabular-nums focus:outline-none focus:ring-2 focus:ring-brand-600"
   415→   415→              autoFocus
   416→   416→            />
   417→   417→            <button
   418→   418→              onClick={handleSave}
   419→   419→              disabled={updateFx.isPending}
   420→   420→              className="inline-flex items-center gap-1 rounded-lg bg-brand-700 px-3 py-1.5 text-xs font-semibold text-white hover:bg-brand-800 disabled:opacity-50 transition-colors"
   421→   421→            >
   422→   422→              {updateFx.isPending ? <Loader2 className="h-3 w-3 animate-spin" /> : <Check className="h-3 w-3" />} Save
   423→   423→            </button>
   424→   424→            <button
   425→   425→              onClick={handleCancel}
   426→   426→              disabled={updateFx.isPending}
   427→   427→              className="inline-flex items-center gap-1 rounded-lg border border-slate-200 px-3 py-1.5 text-xs font-medium text-slate-600 hover:bg-slate-50 disabled:opacity-50 transition-colors"
   428→   428→            >
   429→   429→              Cancel
   430→   430→            </button>
   431→   431→          </div>
   432→   432→        )}
   433→   433→      </div>
   434→   434→    </div>
   435→   435→  )
   436→   436→}
