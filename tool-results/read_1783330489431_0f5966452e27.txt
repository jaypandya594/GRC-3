   266→// ─── Summary Panel ─────────────────────────────────────────────
   267→function SummaryPanel({ selection, pricing, calc }: {
   268→  selection: QuoteBuilderSelection
   269→  pricing: PricingData
   270→  calc: ReturnType<typeof calculateQuote> | null
   271→}) {
   272→  if (!calc) return null
   273→  const selectedFrameworks = pricing.frameworks.filter((f) => selection.selectedFrameworkIds.includes(f.id))
   274→  const tier = pricing.tiers.find((t) => t.id === selection.tierId)
   275→
   276→  return (
   277→    <div className="space-y-3">
   278→      {calc.lines.filter((l) => l.lineType !== 'internal_time').length === 0 && (
   279→        <p className="text-xs text-slate-400 py-4 text-center">
   280→          Select framework(s) and a tier to see pricing
   281→        </p>
   282→      )}
   283→      {selectedFrameworks.length > 0 && (
   284→        <p className="text-xs font-medium text-brand-700">
   285→          {selectedFrameworks.length} framework{selectedFrameworks.length > 1 ? 's' : ''} selected
   286→        </p>
   287→      )}
   288→
   289→      {calc.lines.map((line, i) => {
   290→        const isInternal = line.lineType === 'internal_time'
   291→        const isDiscount = line.lineType === 'discount'
   292→        return (
   293→          <div
   294→            key={i}
   295→            className={cn(
   296→              'flex items-start justify-between gap-2 text-sm',
   297→              isInternal && 'opacity-60',
   298→            )}
   299→          >
   300→            <span className={cn('text-slate-600 flex-1', isInternal && 'italic text-xs', isDiscount && 'text-rose-600')}>
   301→              {line.description}
   302→            </span>
   303→            <span className={cn(
   304→              'font-medium tabular-nums shrink-0',
   305→              isInternal ? 'text-slate-400 text-xs' : isDiscount ? 'text-rose-600' : 'text-slate-900',
   306→            )}>
   307→              {line.amountInr < 0 ? '-' : ''}{formatLineAmount(Math.abs(line.amountInr), calc.billingCurrency, calc.usdInrRate)}
   308→            </span>
   309→          </div>
   310→        )
   311→      })}
   312→
   313→      <div className="border-t border-slate-200 pt-3 space-y-1.5">
   314→        <div className="flex justify-between text-sm">
   315→          <span className="text-slate-600">Subtotal</span>
   316→          <span className="font-medium text-slate-900">{formatLineAmount(calc.subtotalInr, calc.billingCurrency, calc.usdInrRate)}</span>
   317→        </div>
   318→        {calc.discountInr > 0 && (
   319→          <div className="flex justify-between text-sm">
   320→            <span className="text-rose-600">Discount ({calc.discountPct}%)</span>
   321→            <span className="font-medium text-rose-600">-{formatLineAmount(calc.discountInr, calc.billingCurrency, calc.usdInrRate)}</span>
   322→          </div>
   323→        )}
   324→        {calc.gstAmountInr > 0 && (
   325→          <div className="flex justify-between text-sm">
   326→            <span className="text-slate-600">GST @ {calc.gstRate}%</span>
   327→            <span className="font-medium text-slate-900">{formatINR(calc.gstAmountInr)}</span>
   328→          </div>
   329→        )}
   330→        <div className="flex justify-between items-baseline pt-2 border-t border-slate-200">
   331→          <span className="text-sm font-bold text-slate-900">Grand Total</span>
   332→          <div className="text-right">
   333→            {calc.billingCurrency === 'USD' ? (
   334→              <>
   335→                <p className="text-lg font-bold text-brand-700">{formatUSD(calc.totalUsd)}</p>
   336→                <p className="text-xs text-slate-500">No GST</p>
   337→              </>
   338→            ) : (
   339→              <>
   340→                <p className="text-lg font-bold text-brand-700">{formatINR(calc.totalInr)}</p>
   341→                <p className="text-xs text-slate-500">{formatUSD(calc.totalUsd)} @ Rs.{calc.usdInrRate}</p>
   342→              </>
   343→            )}
   344→          </div>
   345→        </div>
   346→        {calc.internalTimeCostInr > 0 && (
   347→          <p className="text-xs text-slate-400 italic pt-1">
   348→            Advisory (not billed): {formatINR(calc.internalTimeCostInr)}
   349→          </p>
   350→        )}
   351→      </div>
   352→    </div>
   353→  )
   354→}

   356→// ─── Step 0: Client + Framework ────────────────────────────────
   357→function FxRateCard({ currentRate }: { currentRate: number }) {
   358→  const queryClient = useQueryClient()
   359→  const [editing, setEditing] = useState(false)
   360→  const [draft, setDraft] = useState(String(currentRate))
   361→
   362→  const updateFx = useMutation({
   363→    mutationFn: (rate: number) => api.put('/admin/fx-rate', { rate }),
   364→    onSuccess: () => {
   365→      queryClient.invalidateQueries({ queryKey: ['pricing'] })
   366→      toast.success('FX rate updated')
   367→      setEditing(false)
   368→    },
   369→    onError: (e: Error) => toast.error(e.message),
   370→  })
   371→
   372→  const handleSave = () => {
   373→    const val = parseFloat(draft)
   374→    if (isNaN(val) || val <= 0) {
   375→      toast.error('Enter a valid positive rate')
   376→      return
   377→    }
   378→    updateFx.mutate(val)
   379→  }
   380→
   381→  const handleCancel = () => {
   382→    setDraft(String(currentRate))
   383→    setEditing(false)
   384→  }
   385→
   386→  return (
   387→    <div className="rounded-xl border border-slate-200 bg-white p-5">
   388→      <div className="flex items-start justify-between gap-4">
   389→        <div className="space-y-1">
   390→          <h3 className="text-sm font-semibold text-slate-900">USD/INR Exchange Rate</h3>
   391→          {!editing && (
   392→            <p className="text-xs text-slate-500">This rate will be used to calculate USD equivalents</p>
   393→          )}
   394→        </div>
   395→        {!editing ? (
   396→          <div className="flex items-center gap-3">
   397→            <span className="text-2xl font-bold text-brand-700 tabular-nums">{currentRate.toFixed(2)}</span>
   398→            <button
   399→              onClick={() => { setDraft(String(currentRate)); setEditing(true) }}
   400→              className="inline-flex items-center gap-1.5 rounded-lg border border-brand-200 bg-brand-50 px-3 py-1.5 text-xs font-medium text-brand-700 hover:bg-brand-100 transition-colors"
   401→            >
   402→              <Pencil className="h-3 w-3" /> Update Rate
   403→            </button>
   404→          </div>
   405→        ) : (
   406→          <div className="flex items-center gap-2">
   407→            <input
   408→              type="number"
   409→              step={0.01}
   410→              min={0.01}
   411→              value={draft}
   412→              onChange={(e) => setDraft(e.target.value)}
   413→              onKeyDown={(e) => { if (e.key === 'Enter') handleSave(); if (e.key === 'Escape') handleCancel() }}
   414→              className="w-28 rounded-lg border border-slate-200 px-2.5 py-1.5 text-sm tabular-nums focus:outline-none focus:ring-2 focus:ring-brand-600"
   415→              autoFocus
   416→            />
   417→            <button
   418→              onClick={handleSave}
   419→              disabled={updateFx.isPending}
   420→              className="inline-flex items-center gap-1 rounded-lg bg-brand-700 px-3 py-1.5 text-xs font-semibold text-white hover:bg-brand-800 disabled:opacity-50 transition-colors"
   421→            >
   422→              {updateFx.isPending ? <Loader2 className="h-3 w-3 animate-spin" /> : <Check className="h-3 w-3" />} Save
   423→            </button>
   424→            <button
   425→              onClick={handleCancel}
   426→              disabled={updateFx.isPending}
   427→              className="inline-flex items-center gap-1 rounded-lg border border-slate-200 px-3 py-1.5 text-xs font-medium text-slate-600 hover:bg-slate-50 disabled:opacity-50 transition-colors"
   428→            >
   429→              Cancel
   430→            </button>
   431→          </div>
   432→        )}
   433→      </div>
   434→    </div>
   435→  )
   436→}

   438→function StepFramework({ selection, update, pricing, clients }: {
   439→  selection: QuoteBuilderSelection
   440→  update: (p: Partial<QuoteBuilderSelection>) => void
   441→  pricing: PricingData
   442→  clients: Client[]
   443→}) {
   444→  const grouped = pricing.frameworks.reduce((acc, fw) => {
   445→    (acc[fw.category] ||= []).push(fw)
   446→    return acc
   447→  }, {} as Record<string, Framework[]>)
   448→
   449→  const toggle = (id: string) => {
   450→    const ids = selection.selectedFrameworkIds
   451→    update({
   452→      selectedFrameworkIds: ids.includes(id)
   453→        ? ids.filter((x) => x !== id)
   454→        : [...ids, id],
   455→    })
   456→  }
   457→
   458→  const clearAll = () => update({ selectedFrameworkIds: [] })
   459→  const selectAll = () => update({ selectedFrameworkIds: pricing.frameworks.map((f) => f.id) })
   460→
   461→  return (
   462→    <div className="space-y-6">
   463→      {/* FX Rate */}
   464→      <FxRateCard currentRate={pricing.fxRate} />
   465→
   466→      {/* Billing Currency */}
   467→      <div className="rounded-xl border border-slate-200 bg-white p-4">
   468→        <div className="flex items-center justify-between">
   469→          <div>
   470→            <p className="text-sm font-semibold text-slate-900">Billing Currency</p>
   471→            <p className="text-xs text-slate-500">USD quotes exclude GST</p>
   472→          </div>
   473→          <div className="flex rounded-lg border border-slate-200 overflow-hidden">
   474→            <button
   475→              onClick={() => update({ billingCurrency: 'INR' })}
   476→              className={cn(
   477→                'px-4 py-1.5 text-xs font-semibold transition-colors',
   478→                selection.billingCurrency === 'INR'
   479→                  ? 'bg-brand-700 text-white'
   480→                  : 'bg-white text-slate-600 hover:bg-slate-50',
   481→              )}
   482→            >
   483→              INR (₹)
   484→            </button>
   485→            <button
   486→              onClick={() => update({ billingCurrency: 'USD' })}
   487→              className={cn(
   488→                'px-4 py-1.5 text-xs font-semibold transition-colors border-l border-slate-200',
   489→                selection.billingCurrency === 'USD'
   490→                  ? 'bg-brand-700 text-white'
   491→                  : 'bg-white text-slate-600 hover:bg-slate-50',
   492→              )}
   493→            >
   494→              USD ($)
   495→            </button>
   496→          </div>
   497→        </div>
   498→      </div>
   499→
   500→      {/* Client selection */}
   501→      <SectionCard title="1. Select Client" description="Choose the prospect this quote is for">
   502→        <select
   503→          value={selection.clientId || ''}
   504→          onChange={(e) => update({ clientId: e.target.value || null })}
   505→          className="w-full rounded-lg border border-slate-200 px-3 py-2.5 text-sm focus:outline-none focus:ring-2 focus:ring-brand-600"
   506→        >
   507→          <option value="">— Select a client —</option>
   508→          {clients.map((c) => (
   509→            <option key={c.id} value={c.id}>
   510→              {c.companyName} ({c.sector || 'N/A'} • {c.companySize})
   511→            </option>
   512→          ))}
   513→        </select>
   514→      </SectionCard>
   515→
   516→      {/* Framework selection — MULTI */}
   517→      <SectionCard
   518→        title="2. Select Framework(s)"
   519→        description="Choose one or more compliance services — all selected frameworks use the same tier"
   520→        action={
   521→          <div className="flex items-center gap-3">
   522→            {selection.selectedFrameworkIds.length > 0 && (
   523→              <button onClick={clearAll} className="text-xs font-medium text-rose-600 hover:text-rose-700">
   524→                Clear All
