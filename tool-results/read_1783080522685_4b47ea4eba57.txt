   266→// ─── Summary Panel ─────────────────────────────────────────────
   267→function SummaryPanel({ selection, pricing, calc }: {
   268→  selection: QuoteBuilderSelection
   269→  pricing: PricingData
   270→  calc: ReturnType<typeof calculateQuote> | null
   271→}) {
   272→  if (!calc) return null
   273→  const selectedFrameworks = pricing.frameworks.filter((f) => selection.selectedFrameworkIds.includes(f.id))
   274→  const tier = pricing.tiers.find((t) => t.id === selection.tierId)
   275→
   276→  return (
   277→    <div className="space-y-3">
   278→      {calc.lines.filter((l) => l.lineType !== 'internal_time').length === 0 && (
   279→        <p className="text-xs text-slate-400 py-4 text-center">
   280→          Select framework(s) and a tier to see pricing
   281→        </p>
   282→      )}
   283→      {selectedFrameworks.length > 0 && (
   284→        <p className="text-xs font-medium text-brand-700">
   285→          {selectedFrameworks.length} framework{selectedFrameworks.length > 1 ? 's' : ''} selected
   286→        </p>
   287→      )}
   288→
   289→      {calc.lines.map((line, i) => {
   290→        const isInternal = line.lineType === 'internal_time'
   291→        const isDiscount = line.lineType === 'discount'
   292→        return (
   293→          <div
   294→            key={i}
   295→            className={cn(
   296→              'flex items-start justify-between gap-2 text-sm',
   297→              isInternal && 'opacity-60',
   298→            )}
   299→          >
   300→            <span className={cn('text-slate-600 flex-1', isInternal && 'italic text-xs', isDiscount && 'text-rose-600')}>
   301→              {line.description}
   302→            </span>
   303→            <span className={cn(
   304→              'font-medium tabular-nums shrink-0',
   305→              isInternal ? 'text-slate-400 text-xs' : isDiscount ? 'text-rose-600' : 'text-slate-900',
   306→            )}>
   307→              {line.amountInr < 0 ? '-' : ''}{formatINR(Math.abs(line.amountInr))}
   308→            </span>
   309→          </div>
   310→        )
   311→      })}
   312→
   313→      <div className="border-t border-slate-200 pt-3 space-y-1.5">
   314→        <div className="flex justify-between text-sm">
   315→          <span className="text-slate-600">Subtotal</span>
   316→          <span className="font-medium text-slate-900">{formatINR(calc.subtotalInr)}</span>
   317→        </div>
   318→        {calc.discountInr > 0 && (
   319→          <div className="flex justify-between text-sm">
   320→            <span className="text-rose-600">Discount ({calc.discountPct}%)</span>
   321→            <span className="font-medium text-rose-600">-{formatINR(calc.discountInr)}</span>
   322→          </div>
   323→        )}
   324→        <div className="flex justify-between text-sm">
   325→          <span className="text-slate-600">GST @ {calc.gstRate}%</span>
   326→          <span className="font-medium text-slate-900">{formatINR(calc.gstAmountInr)}</span>
   327→        </div>
   328→        <div className="flex justify-between items-baseline pt-2 border-t border-slate-200">
   329→          <span className="text-sm font-bold text-slate-900">Grand Total</span>
   330→          <div className="text-right">
   331→            <p className="text-lg font-bold text-brand-700">{formatINR(calc.totalInr)}</p>
   332→            <p className="text-xs text-slate-500">{formatUSD(calc.totalUsd)} @ ₹{calc.usdInrRate}</p>
   333→          </div>
   334→        </div>
   335→        {calc.internalTimeCostInr > 0 && (
   336→          <p className="text-xs text-slate-400 italic pt-1">
   337→            Advisory (not billed): {formatINR(calc.internalTimeCostInr)}
   338→          </p>
   339→        )}
   340→      </div>
   341→    </div>
   342→  )
   343→}

   345→// ─── Step 0: Client + Framework ────────────────────────────────
   346→function FxRateCard({ currentRate }: { currentRate: number }) {
   347→  const queryClient = useQueryClient()
   348→  const [editing, setEditing] = useState(false)
   349→  const [draft, setDraft] = useState(String(currentRate))
   350→
   351→  const updateFx = useMutation({
   352→    mutationFn: (rate: number) => api.put('/admin/fx-rate', { rate }),
   353→    onSuccess: () => {
   354→      queryClient.invalidateQueries({ queryKey: ['pricing'] })
   355→      toast.success('FX rate updated')
   356→      setEditing(false)
   357→    },
   358→    onError: (e: Error) => toast.error(e.message),
   359→  })
   360→
   361→  const handleSave = () => {
   362→    const val = parseFloat(draft)
   363→    if (isNaN(val) || val <= 0) {
   364→      toast.error('Enter a valid positive rate')
   365→      return
   366→    }
   367→    updateFx.mutate(val)
   368→  }
   369→
   370→  const handleCancel = () => {
   371→    setDraft(String(currentRate))
   372→    setEditing(false)
   373→  }
   374→
   375→  return (
   376→    <div className="rounded-xl border border-slate-200 bg-white p-5">
   377→      <div className="flex items-start justify-between gap-4">
   378→        <div className="space-y-1">
   379→          <h3 className="text-sm font-semibold text-slate-900">USD/INR Exchange Rate</h3>
   380→          {!editing && (
   381→            <p className="text-xs text-slate-500">This rate will be used to calculate USD equivalents</p>
   382→          )}
   383→        </div>
   384→        {!editing ? (
   385→          <div className="flex items-center gap-3">
   386→            <span className="text-2xl font-bold text-brand-700 tabular-nums">{currentRate.toFixed(2)}</span>
   387→            <button
   388→              onClick={() => { setDraft(String(currentRate)); setEditing(true) }}
   389→              className="inline-flex items-center gap-1.5 rounded-lg border border-brand-200 bg-brand-50 px-3 py-1.5 text-xs font-medium text-brand-700 hover:bg-brand-100 transition-colors"
   390→            >
   391→              <Pencil className="h-3 w-3" /> Update Rate
   392→            </button>
   393→          </div>
   394→        ) : (
   395→          <div className="flex items-center gap-2">
   396→            <input
   397→              type="number"
   398→              step={0.01}
   399→              min={0.01}
   400→              value={draft}
   401→              onChange={(e) => setDraft(e.target.value)}
   402→              onKeyDown={(e) => { if (e.key === 'Enter') handleSave(); if (e.key === 'Escape') handleCancel() }}
   403→              className="w-28 rounded-lg border border-slate-200 px-2.5 py-1.5 text-sm tabular-nums focus:outline-none focus:ring-2 focus:ring-brand-600"
   404→              autoFocus
   405→            />
   406→            <button
   407→              onClick={handleSave}
   408→              disabled={updateFx.isPending}
   409→              className="inline-flex items-center gap-1 rounded-lg bg-brand-700 px-3 py-1.5 text-xs font-semibold text-white hover:bg-brand-800 disabled:opacity-50 transition-colors"
   410→            >
   411→              {updateFx.isPending ? <Loader2 className="h-3 w-3 animate-spin" /> : <Check className="h-3 w-3" />} Save
   412→            </button>
   413→            <button
   414→              onClick={handleCancel}
   415→              disabled={updateFx.isPending}
   416→              className="inline-flex items-center gap-1 rounded-lg border border-slate-200 px-3 py-1.5 text-xs font-medium text-slate-600 hover:bg-slate-50 disabled:opacity-50 transition-colors"
   417→            >
   418→              Cancel
   419→            </button>
   420→          </div>
   421→        )}
   422→      </div>
   423→    </div>
   424→  )
   425→}

   427→function StepFramework({ selection, update, pricing, clients }: {
   428→  selection: QuoteBuilderSelection
   429→  update: (p: Partial<QuoteBuilderSelection>) => void
   430→  pricing: PricingData
   431→  clients: Client[]
   432→}) {
   433→  const grouped = pricing.frameworks.reduce((acc, fw) => {
   434→    (acc[fw.category] ||= []).push(fw)
   435→    return acc
   436→  }, {} as Record<string, Framework[]>)
   437→
   438→  const toggle = (id: string) => {
   439→    const ids = selection.selectedFrameworkIds
   440→    update({
   441→      selectedFrameworkIds: ids.includes(id)
   442→        ? ids.filter((x) => x !== id)
   443→        : [...ids, id],
   444→    })
   445→  }
   446→
   447→  const clearAll = () => update({ selectedFrameworkIds: [] })
   448→  const selectAll = () => update({ selectedFrameworkIds: pricing.frameworks.map((f) => f.id) })
   449→
   450→  return (
   451→    <div className="space-y-6">
   452→      {/* FX Rate */}
   453→      <FxRateCard currentRate={pricing.fxRate} />
   454→
   455→      {/* Client selection */}
   456→      <SectionCard title="1. Select Client" description="Choose the prospect this quote is for">
   457→        <select
   458→          value={selection.clientId || ''}
   459→          onChange={(e) => update({ clientId: e.target.value || null })}
   460→          className="w-full rounded-lg border border-slate-200 px-3 py-2.5 text-sm focus:outline-none focus:ring-2 focus:ring-brand-600"
   461→        >
   462→          <option value="">— Select a client —</option>
   463→          {clients.map((c) => (
   464→            <option key={c.id} value={c.id}>
   465→              {c.companyName} ({c.sector || 'N/A'} • {c.companySize})
   466→            </option>
   467→          ))}
   468→        </select>
   469→      </SectionCard>
   470→
   471→      {/* Framework selection — MULTI */}
   472→      <SectionCard
   473→        title="2. Select Framework(s)"
   474→        description="Choose one or more compliance services — all selected frameworks use the same tier"
   475→        action={
   476→          <div className="flex items-center gap-3">
   477→            {selection.selectedFrameworkIds.length > 0 && (
   478→              <button onClick={clearAll} className="text-xs font-medium text-rose-600 hover:text-rose-700">
   479→                Clear All
   480→              </button>
   481→            )}
   482→            {selection.selectedFrameworkIds.length < pricing.frameworks.length && (
   483→              <button onClick={selectAll} className="text-xs font-medium text-brand-700 hover:text-brand-800">
   484→                Select All
   485→              </button>
   486→            )}
   487→          </div>
   488→        }
   489→      >
   490→        {selection.selectedFrameworkIds.length > 0 && (
   491→          <div className="mb-4 rounded-lg bg-brand-50 border border-brand-200 px-3 py-2">
   492→            <p className="text-xs font-medium text-brand-700">
   493→              <Layers className="inline h-3.5 w-3.5 mr-1" />
   494→              {selection.selectedFrameworkIds.length} framework{selection.selectedFrameworkIds.length > 1 ? 's' : ''} selected
   495→            </p>
   496→          </div>
   497→        )}
   498→        {Object.entries(grouped).map(([category, fws]) => {
   499→          const meta = FRAMEWORK_CATEGORY_META[category] || { label: category, color: 'slate' }
   500→          return (
   501→            <div key={category} className="mb-6 last:mb-0">
   502→              <h4 className="text-xs font-bold uppercase tracking-wide text-slate-500 mb-3">{meta.label}</h4>
   503→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">
   504→                {fws.map((fw) => (
   505→                  <PricingCard
   506→                    key={fw.id}
   507→                    selected={selection.selectedFrameworkIds.includes(fw.id)}
   508→                    onClick={() => toggle(fw.id)}
   509→                    title={fw.name}
   510→                    description={fw.description || undefined}
   511→                    badge={meta.label}
   512→                  />
   513→                ))}
   514→              </div>
   515→            </div>
   516→          )
   517→        })}
   518→      </SectionCard>
   519→    </div>
   520→  )
   521→}

   523→// ─── Step 1: Tier ──────────────────────────────────────────────
   524→function StepTier({ selection, update, pricing }: {
   525→  selection: QuoteBuilderSelection
   526→  update: (p: Partial<QuoteBuilderSelection>) => void
   527→  pricing: PricingData
   528→}) {

   531→    return (
   532→      <SectionCard title="Select Tier">

   534→      </SectionCard>
   535→    )
   536→  }

   538→  return (

   550→              (p) => p.frameworkId === fw.id && p.tierId === tier.id,

   563→              subtitle={`Retainer: ${tier.retainerPct}%`}
   564→              fee={totalFee}
   565→              feeLabel="total project fee"
   566→              description={totalRetainer > 0 ? `+ ${formatINR(totalRetainer)} annual retainer` : undefined}
   567→            />
   568→          )
   569→        })}
   570→      </div>
   571→
   572→      {selection.tierId && (
   573→        <div className="rounded-lg border border-slate-200 p-4 bg-slate-50">
   574→          <label className="flex items-center gap-3 cursor-pointer">
   575→            <input
   576→              type="checkbox"
   577→              checked={selection.includeRetainer}
   578→              onChange={(e) => update({ includeRetainer: e.target.checked })}

   580→            />
   581→            <div>
   582→              <p className="text-sm font-medium text-slate-900">Include Annual Retainer</p>
   583→              <p className="text-xs text-slate-500">
   584→                Adds retainer line items for each selected framework ({pricing.tiers.find((t) => t.id === selection.tierId)?.retainerPct}% of each project fee)

   594→// ─── Step 2: Auditor Fees ──────────────────────────────────────

   606→  const toggle = (id: string) => {

   622→        selection.selectedAuditorFeeIds.length > 0 && (

   624→            Clear All
   625→          </button>
   626→        )

   632→            <h4 className="text-xs font-bold uppercase tracking-wide text-slate-500 mb-2">{standard}</h4>

   635→                <PricingCard
   636→                  key={fee.id}
   637→                  selected={selection.selectedAuditorFeeIds.includes(fee.id)}
   638→                  onClick={() => toggle(fee.id)}

   644→            </div>
   645→          </div>
   646→        ))}
   647→      </div>
   648→    </SectionCard>
   649→  )
