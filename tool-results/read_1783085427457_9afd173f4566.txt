   536→   536→// ─── Animated Counter ──────────────────────────────────────────
   537→   537→function AnimatedCounter({ value, duration = 1200 }: { value: number; duration?: number }) {
   538→   538→  const [display, setDisplay] = useState(0)
   539→   539→  const prevValue = useRef(0)
   540→   540→
   541→   541→  useEffect(() => {
   542→   542→    const start = prevValue.current
   543→   543→    const end = value
   544→   544→    const startTime = performance.now()
   545→   545→
   546→   546→    const animate = (currentTime: number) => {
   547→   547→      const elapsed = currentTime - startTime
   548→   548→      const progress = Math.min(elapsed / duration, 1)
   549→   549→      // Ease out cubic
   550→   550→      const eased = 1 - Math.pow(1 - progress, 3)
   551→   551→      setDisplay(Math.round(start + (end - start) * eased))
   552→   552→      if (progress < 1) requestAnimationFrame(animate)
   553→   553→    }
   554→   554→
   555→   555→    requestAnimationFrame(animate)
   556→   556→    prevValue.current = value
   557→   557→  }, [value, duration])
   558→   558→
   559→   559→  return <>{formatNumberIN(display)}</>
   560→   560→}

   563→   563→function exportCsv() {
   564→   564→  const rows = [
   565→   565→    ['Client', 'Framework', 'Tier', 'Status', 'Subtotal (INR)', 'Discount (INR)', 'GST (INR)', 'Total (INR)', 'USD', 'Created'],
   566→   566→  ]
   567→   567→
   568→   568→  // We'll fetch fresh data for export
   569→   569→  fetch('/api/quotes?status=all')
   570→   570→    .then((r) => r.json())
   571→   571→    .then((res) => {
   572→   572→      const quotes = res.data || []
   573→   573→      for (const q of quotes) {
   574→   574→        rows.push([
   575→   575→          q.client?.companyName || '',
   576→   576→          getQuoteFrameworkString(q) || '',
   577→   577→          q.tier?.name || '',
   578→   578→          q.status,
   579→   579→          String(q.subtotalInr),
   580→   580→          String(q.discountInr),
   581→   581→          String(q.gstAmountInr),
   582→   582→          String(q.totalInr),
   583→   583→          String(q.totalUsd),
   584→   584→          new Date(q.createdAt).toLocaleDateString('en-IN'),
   585→   585→        ])
   586→   586→      }
   587→   587→      const csv = rows.map((r) => r.map((c) => `"${c}"`).join(',')).join('\n')
   588→   588→      const blob = new Blob([csv], { type: 'text/csv' })
   589→   589→      const url = URL.createObjectURL(blob)
   590→   590→      const a = document.createElement('a')
   591→   591→      a.href = url
   592→   592→      a.download = `isecurify-quotes-${new Date().toISOString().split('T')[0]}.csv`
   593→   593→      document.body.appendChild(a)
   594→   594→      a.click()
   595→   595→      document.body.removeChild(a)
   596→   596→      URL.revokeObjectURL(url)
   597→   597→    })
   598→   598→    .catch(() => {
   599→   599→      // fallback: ignore
   600→   600→    })
   601→   601→}
