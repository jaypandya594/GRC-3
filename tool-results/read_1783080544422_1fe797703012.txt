    55→    55→async function downloadQuotePdf(id: string): Promise<void> {
    56→    56→  try {
    57→    57→    // Primary: open PDF in new tab — works in sandbox/preview/iframe environments
    58→    58→    // The server sends Content-Disposition: attachment so browser will download it
    59→    59→    const win = window.open(`/api/quote-pdf/${id}`, '_blank')
    60→    60→    if (!win) {
    61→    61→      // Popup blocked — fall back to blob download
    62→    62→      const res = await fetch(`/api/quote-pdf/${id}`)
    63→    63→      if (!res.ok) throw new Error(`Server returned ${res.status}`)
    64→    64→      const blob = await res.blob()
    65→    65→      const blobUrl = URL.createObjectURL(blob)
    66→    66→      const a = document.createElement('a')
    67→    67→      a.style.display = 'none'
    68→    68→      a.href = blobUrl
    69→    69→      a.download = `iSecurify-Quote-${id.slice(-8).toUpperCase()}.pdf`
    70→    70→      document.body.appendChild(a)
    71→    71→      a.click()
    72→    72→      setTimeout(() => {
    73→    73→        document.body.removeChild(a)
    74→    74→        URL.revokeObjectURL(blobUrl)
    75→    75→      }, 5000)
    76→    76→    }
    77→    77→  } catch (err) {
    78→    78→    console.error('[PDF Download] Error:', err)
    79→    79→    throw err
    80→    80→  }
    81→    81→}

   730→   730→function getFrameworkNamesFromLines(quote: Quote): string {
   731→   731→  const consultingLines = (quote.lineItems || []).filter((l) => l.lineType === 'consulting_fee')
   732→   732→  if (consultingLines.length > 1) {
   733→   733→    return consultingLines
   734→   734→      .map((l) => l.description.replace(/ — Consulting Fee.*/, '').trim())
   735→   735→      .join(', ')
   736→   736→  }
   737→   737→  if (consultingLines.length === 1) {
   738→   738→    return consultingLines[0].description.replace(/ — Consulting Fee.*/, '').trim()
   739→   739→  }
   740→   740→  return quote.framework?.name || '—'
   741→   741→}
