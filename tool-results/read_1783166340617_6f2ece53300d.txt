   223→   223→const FRAMEWORK_CATEGORIES: FrameworkCategory[] = ['IT_Security', 'Privacy', 'Quality', 'Food_Safety', 'Industry']
   224→   224→
   225→   225→function FrameworksSection({ frameworks }: { frameworks: Framework[] }) {
   226→   226→  const qc = useQueryClient()
   227→   227→  const [open, setOpen] = useState(false)
   228→   228→  const [editing, setEditing] = useState<Framework | null>(null)
   229→   229→
   230→   230→  const [form, setForm] = useState({ name: '', category: 'IT_Security' as FrameworkCategory, description: '', sortOrder: 0, isActive: true })
   231→   231→
   232→   232→  const resetForm = useCallback((fw?: Framework) => {
   233→   233→    if (fw) {
   234→   234→      setForm({ name: fw.name, category: fw.category, description: fw.description ?? '', sortOrder: fw.sortOrder, isActive: fw.isActive })
   235→   235→    } else {
   236→   236→      setForm({ name: '', category: 'IT_Security', description: '', sortOrder: 0, isActive: true })
   237→   237→    }
   238→   238→  }, [])
   239→   239→
   240→   240→  const openAdd = () => { resetForm(); setEditing(null); setOpen(true) }
   241→   241→  const openEdit = (fw: Framework) => { resetForm(fw); setEditing(fw); setOpen(true) }
   242→   242→
   243→   243→  const addMutation = useMutation({
   244→   244→    mutationFn: (body: typeof form) => api.post<Framework>('/admin/frameworks', body),
   245→   245→    onSuccess: () => { toast.success('Framework added'); setOpen(false); qc.invalidateQueries({ queryKey: ['pricing'] }) },
   246→   246→    onError: (e: Error) => toast.error(e.message),
   247→   247→  })
   248→   248→
   249→   249→  const editMutation = useMutation({
   250→   250→    mutationFn: (body: Partial<typeof form>) => api.put<Framework>(`/admin/frameworks/${editing!.id}`, body),
   251→   251→    onSuccess: () => { toast.success('Framework updated'); setOpen(false); qc.invalidateQueries({ queryKey: ['pricing'] }) },
   252→   252→    onError: (e: Error) => toast.error(e.message),
   253→   253→  })
   254→   254→
   255→   255→  const deleteMutation = useMutation({
   256→   256→    mutationFn: (id: string) => api.delete(`/admin/frameworks/${id}`),
   257→   257→    onSuccess: () => { toast.success('Framework deleted'); qc.invalidateQueries({ queryKey: ['pricing'] }) },
   258→   258→    onError: (e: Error) => toast.error(e.message),
   259→   259→  })
   260→   260→
   261→   261→  const handleSubmit = () => {
   262→   262→    if (!form.name.trim()) { toast.error('Name is required'); return }
   263→   263→    if (editing) editMutation.mutate(form)
   264→   264→    else addMutation.mutate(form)
   265→   265→  }
   266→   266→
   267→   267→  const grouped = frameworks.reduce((acc, fw) => {
   268→   268→    (acc[fw.category] ||= []).push(fw)
   269→   269→    return acc
   270→   270→  }, {} as Record<string, Framework[]>)
   271→   271→
   272→   272→  return (
   273→   273→    <SectionCard
   274→   274→      title="Frameworks"
   275→   275→      description={`${frameworks.length} frameworks, grouped by category`}
   276→   276→      action={<Button size="sm" className="bg-brand-700 hover:bg-brand-800 text-white" onClick={openAdd}><Plus className="h-4 w-4 mr-1" /> Add Framework</Button>}
   277→   277→    >
   278→   278→      {Object.keys(grouped).length === 0 ? (
   279→   279→        <EmptyState title="No frameworks" description="Add your first framework to get started" />
   280→   280→      ) : (
   281→   281→        <div className="space-y-5">
   282→   282→          {Object.entries(grouped).map(([category, fws]) => {
   283→   283→            const meta = FRAMEWORK_CATEGORY_META[category] || { label: category }
   284→   284→            return (
   285→   285→              <div key={category}>
   286→   286→                <h4 className="text-xs font-bold uppercase tracking-wide text-slate-500 mb-2">{meta.label}</h4>
   287→   287→                <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-2">
   288→   288→                  {fws.map((fw) => (
   289→   289→                    <div key={fw.id} className="group relative rounded-lg border border-slate-200 p-3 hover:border-brand-300 transition-colors">
   290→   290→                      <div className="absolute top-2 right-2 flex gap-0.5 opacity-0 group-hover:opacity-100 transition-opacity">
   291→   291→                        <EditButton onClick={() => openEdit(fw)} />
   292→   292→                        <DeleteConfirm label={fw.name} onConfirm={() => deleteMutation.mutate(fw.id)} />
   293→   293→                      </div>
   294→   294→                      <p className="text-sm font-medium text-slate-900">{fw.name}</p>
   295→   295→                      <p className="text-xs text-slate-500 line-clamp-2 mt-0.5">{fw.description}</p>
   296→   296→                      {!fw.isActive && <span className="inline-block mt-1 px-1.5 py-0.5 rounded text-[10px] font-semibold bg-slate-100 text-slate-500">Inactive</span>}
   297→   297→                    </div>
   298→   298→                  ))}
   299→   299→                </div>
   300→   300→              </div>
   301→   301→            )
   302→   302→          })}
   303→   303→        </div>
   304→   304→      )}
   305→   305→
   306→   306→      <Dialog open={open} onOpenChange={(v) => { setOpen(v); if (!v) setEditing(null) }}>
   307→   307→        <DialogContent className="sm:max-w-md">
   308→   308→          <DialogHeader>
   309→   309→            <DialogTitle>{editing ? 'Edit Framework' : 'Add Framework'}</DialogTitle>
   310→   310→            <DialogDescription>{editing ? 'Update framework details' : 'Create a new certification framework'}</DialogDescription>
   311→   311→          </DialogHeader>
   312→   312→          <div className="space-y-4 pt-2">
   313→   313→            <FormField label="Name">
   314→   314→              <Input value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} placeholder="e.g. ISO 27001" />
   315→   315→            </FormField>
   316→   316→            <FormField label="Category">
   317→   317→              <Select value={form.category} onValueChange={(v) => setForm({ ...form, category: v as FrameworkCategory })}>
   318→   318→                <SelectTrigger><SelectValue /></SelectTrigger>
   319→   319→                <SelectContent>
   320→   320→                  {FRAMEWORK_CATEGORIES.map((c) => (
   321→   321→                    <SelectItem key={c} value={c}>{FRAMEWORK_CATEGORY_META[c]?.label ?? c}</SelectItem>
   322→   322→                  ))}
   323→   323→                </SelectContent>
   324→   324→              </Select>
   325→   325→            </FormField>
   326→   326→            <FormField label="Description">
   327→   327→              <Textarea value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} placeholder="Brief description..." rows={2} />
   328→   328→            </FormField>
   329→   329→            <FormField label="Sort Order">
   330→   330→              <Input type="number" value={form.sortOrder} onChange={(e) => setForm({ ...form, sortOrder: parseInt(e.target.value) || 0 })} />
   331→   331→            </FormField>
   332→   332→            <div className="flex items-center gap-2">
   333→   333→              <Switch checked={form.isActive} onCheckedChange={(v) => setForm({ ...form, isActive: v })} />
   334→   334→              <Label className="text-xs text-slate-600">Active</Label>
   335→   335→            </div>
   336→   336→            <div className="flex justify-end gap-2 pt-2">
   337→   337→              <Button variant="outline" size="sm" onClick={() => setOpen(false)}>Cancel</Button>
   338→   338→              <Button size="sm" className="bg-brand-700 hover:bg-brand-800 text-white" onClick={handleSubmit} disabled={addMutation.isPending || editMutation.isPending}>
   339→   339→                {(addMutation.isPending || editMutation.isPending) && <Loader2 className="h-3.5 w-3.5 mr-1 animate-spin" />}
   340→   340→                {editing ? 'Update' : 'Create'}
   341→   341→              </Button>
   342→   342→            </div>
   343→   343→          </div>
   344→   344→        </DialogContent>
   345→   345→      </Dialog>
   346→   346→    </SectionCard>
   347→   347→  )
   348→   348→}
