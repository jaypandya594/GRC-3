    59→    59→async function downloadQuotePdf(id: string): Promise<void> {
    60→    60→  try {
    61→    61→    const win = window.open(`/api/quote-pdf/${id}`, '_blank')
    62→    62→    if (!win) {
    63→    63→      const res = await fetch(`/api/quote-pdf/${id}`)
    64→    64→      if (!res.ok) throw new Error(`Server returned ${res.status}`)
    65→    65→      const blob = await res.blob()
    66→    66→      const blobUrl = URL.createObjectURL(blob)
    67→    67→      const a = document.createElement('a')
    68→    68→      a.style.display = 'none'
    69→    69→      a.href = blobUrl
    70→    70→      a.download = `iSecurify-Quote-${id.slice(-8).toUpperCase()}.pdf`
    71→    71→      document.body.appendChild(a)
    72→    72→      a.click()
    73→    73→      setTimeout(() => {
    74→    74→        document.body.removeChild(a)
    75→    75→        URL.revokeObjectURL(blobUrl)
    76→    76→      }, 5000)
    77→    77→    }
    78→    78→  } catch (err) {
    79→    79→    console.error('[PDF Download] Error:', err)
    80→    80→    throw err
    81→    81→  }
    82→    82→}

    86→    86→/** Get all framework names from the QuoteFramework join table, falling back to line items, then single framework */
    87→    87→function getFrameworkDisplayNames(quote: Quote): string[] {
    88→    88→  // 1. Prefer the join table
    89→    89→  if (quote.quoteFrameworks && quote.quoteFrameworks.length > 0) {
    90→    90→    return quote.quoteFrameworks
    91→    91→      .sort((a, b) => a.sortOrder - b.sortOrder)
    92→    92→      .map((qf) => qf.framework?.name || 'Unknown')
    93→    93→  }
    94→    94→  // 2. Fall back to line items
    95→    95→  const consultingLines = (quote.lineItems || []).filter((l) => l.lineType === 'consulting_fee')
    96→    96→  if (consultingLines.length > 0) {
    97→    97→    return consultingLines.map((l) => l.description.replace(/ — Consulting Fee.*/, '').trim())
    98→    98→  }
    99→    99→  // 3. Last resort: single framework field
   100→   100→  if (quote.framework?.name) return [quote.framework.name]
   101→   101→  return []
   102→   102→}
   103→   103→
   104→   104→function getFrameworkDisplayString(quote: Quote): string {
   105→   105→  return getFrameworkDisplayNames(quote).join(' + ') || '—'
   106→   106→}
   107→   107→
   108→   108→function FrameworkBadges({ quote }: { quote: Quote }) {
   109→   109→  const names = getFrameworkDisplayNames(quote)
   110→   110→  if (names.length === 0) return <span className="text-slate-400">—</span>
   111→   111→  if (names.length <= 2) {
   112→   112→    return <span className="text-slate-600">{names.join(' + ')}</span>
   113→   113→  }
   114→   114→  // Show first 2 + count for long lists
   115→   115→  return (
   116→   116→    <span className="text-slate-600">
   117→   117→      {names.slice(0, 2).join(' + ')} <span className="text-slate-400">+{names.length - 2} more</span>
   118→   118→    </span>
   119→   119→  )
   120→   120→}
