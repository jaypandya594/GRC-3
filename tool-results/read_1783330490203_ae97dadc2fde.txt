    55→async function downloadQuotePdf(id: string): Promise<void> {
    56→  try {
    57→    // Primary: open PDF in new tab — works in sandbox/preview/iframe environments
    58→    // The server sends Content-Disposition: attachment so browser will download it
    59→    const win = window.open(`/api/quote-pdf/${id}`, '_blank')
    60→    if (!win) {
    61→      // Popup blocked — fall back to blob download
    62→      const res = await fetch(`/api/quote-pdf/${id}`)
    63→      if (!res.ok) throw new Error(`Server returned ${res.status}`)
    64→      const blob = await res.blob()
    65→      const blobUrl = URL.createObjectURL(blob)
    66→      const a = document.createElement('a')
    67→      a.style.display = 'none'
    68→      a.href = blobUrl
    69→      a.download = `iSecurify-Quote-${id.slice(-8).toUpperCase()}.pdf`
    70→      document.body.appendChild(a)
    71→      a.click()
    72→      setTimeout(() => {
    73→        document.body.removeChild(a)
    74→        URL.revokeObjectURL(blobUrl)
    75→      }, 5000)
    76→    }
    77→  } catch (err) {
    78→    console.error('[PDF Download] Error:', err)
    79→    throw err
    80→  }
    81→}

   730→function getFrameworkNamesFromLines(quote: Quote): string {
   731→  const consultingLines = (quote.lineItems || []).filter((l) => l.lineType === 'consulting_fee')
   732→  if (consultingLines.length > 1) {
   733→    return consultingLines
   734→      .map((l) => l.description.replace(/ — Consulting Fee.*/, '').trim())
   735→      .join(', ')
   736→  }
   737→  if (consultingLines.length === 1) {
   738→    return consultingLines[0].description.replace(/ — Consulting Fee.*/, '').trim()
   739→  }
   740→  return quote.framework?.name || '—'
   741→}
   742→
   743→function getAllFrameworkNames(quote: Quote): string {
   744→  const consultingLines = (quote.lineItems || []).filter((l) => l.lineType === 'consulting_fee')
   745→  if (consultingLines.length > 0) {
   746→    return consultingLines
   747→      .map((l) => l.description.replace(/ — Consulting Fee.*/, '').trim())
   748→      .join(', ')
   749→  }
   750→  return quote.framework?.name || '—'
   751→}

   753→function buildQuoteSubtitle(quote: Quote): string {
   754→  const fwNames = getAllFrameworkNames(quote)
   755→  // Truncate if too long
   756→  const fwDisplay = fwNames.length > 50 ? fwNames.slice(0, 47) + '...' : fwNames
   757→  return `${fwDisplay} • ${quote.tier?.name || '—'} • v${quote.version}`
   758→}
