    56→  tiers: Tier[]
    57→  frameworkPrices: Array<{ id: string; frameworkId: string; tierId: string; projectFeeInr: number; retainerFeeInr: number }>

    65→const STEP_LABELS = ['Framework', 'Tier', 'Auditor', 'Add-ons', 'Tools & DPO', 'Review']
    66→
    67→export function QuoteBuilderView() {
    68→  const store = useQuoteBuilderStore()
    69→  // Defensive: ensure arrays are always arrays (stale localStorage guard)
    70→  const selection: QuoteBuilderSelection = {
    71→    ...DEFAULT_QUOTE_BUILDER_SELECTION,
    72→    ...store.selection,
    73→    selectedFrameworkIds: Array.isArray(store.selection?.selectedFrameworkIds)
    74→      ? store.selection.selectedFrameworkIds
    75→      : [],
    76→    selectedAuditorFeeIds: Array.isArray(store.selection?.selectedAuditorFeeIds)
    77→      ? store.selection.selectedAuditorFeeIds
    78→      : [],
    79→    selectedAddonIds: Array.isArray(store.selection?.selectedAddonIds)
    80→      ? store.selection.selectedAddonIds
    81→      : [],
    82→  }
    83→  const { currentStep, setStep, nextStep, prevStep, update, reset, editingQuoteId, frameworkChangedMidWizard } = store
    84→  const { navigate } = useNavStore()
    85→
    86→  const { data: pricing, isLoading } = useQuery({
    87→    queryKey: ['pricing'],
    88→    queryFn: () => api.get<PricingData>('/pricing'),
    89→  })
    90→
    91→  const { data: clients } = useQuery({
    92→    queryKey: ['clients'],
    93→    queryFn: () => api.get<Client[]>('/clients'),
    94→  })
    95→
    96→  const calc = pricing
    97→    ? calculateQuote(selection, {
    98→        frameworks: pricing.frameworks,
    99→        tiers: pricing.tiers,
   100→        frameworkPrices: pricing.frameworkPrices,
   101→        auditorFees: pricing.auditorFees,
   102→        addonServices: pricing.addonServices,
   103→        grcTools: pricing.grcTools,
   104→        dpoVcisoPackages: pricing.dpoVcisoPackages,
   105→        usdInrRate: pricing.fxRate,
   106→      })
   107→    : null
   108→
   109→  const qc = useQueryClient()
   110→
   111→  // Cascading validation: when frameworks change mid-wizard, filter stale auditor fees
   112→  useEffect(() => {
   113→    if (frameworkChangedMidWizard && pricing && currentStep > 0) {
   114→      // Get framework names for selected frameworks
   115→      const selectedFwNames = new Set(
   116→        pricing.frameworks
   117→          .filter((f) => selection.selectedFrameworkIds.includes(f.id))
   118→          .map((f) => f.name),
   119→      )
   120→      // Filter auditor fee IDs: keep only those whose standardName matches a selected framework
   121→      const validAuditorIds = pricing.auditorFees
   122→        .filter((af) => selectedFwNames.has(af.standardName))
   123→        .map((af) => af.id)
   124→      const staleIds = selection.selectedAuditorFeeIds.filter((id) => !validAuditorIds.includes(id))
   125→      if (staleIds.length > 0) {
   126→        update({
   127→          selectedAuditorFeeIds: selection.selectedAuditorFeeIds.filter((id) => validAuditorIds.includes(id)),
   128→        })
   129→      }
   130→    }
   131→  }, [frameworkChangedMidWizard, pricing, currentStep, selection.selectedFrameworkIds, selection.selectedAuditorFeeIds, update])
   132→
   133→  const handleSetStep = (step: number) => {
   134→    setStep(step)
   135→    // Clear the framework change warning when user acknowledges by navigating
   136→    if (frameworkChangedMidWizard) {
   137→      useQuoteBuilderStore.setState({ frameworkChangedMidWizard: false, prevFrameworkIds: selection.selectedFrameworkIds })
   138→    }
   139→  }
   140→
   141→  const saveMutation = useMutation({
   142→    mutationFn: async (vars: { submit: boolean }) => {
   143→      if (editingQuoteId) {
   144→        return api.put(`/quotes/${editingQuoteId}`, { selection, submit: vars.submit })
   145→      }
   146→      return api.post('/quotes', { selection, submit: vars.submit })
   147→    },
   148→    onSuccess: (quote, vars) => {
   149→      toast.success(editingQuoteId
   150→        ? (vars.submit ? 'Quote updated & submitted for review' : 'Quote updated')
   151→        : (vars.submit ? 'Quote submitted for review' : 'Draft saved'))
   152→      qc.invalidateQueries({ queryKey: ['quotes'] })
   153→      qc.invalidateQueries({ queryKey: ['dashboard'] })
   154→      reset()
   155→      navigate('quotes', { id: (quote as { id: string }).id })
   156→    },
   157→    onError: (e: Error) => toast.error(e.message),
   158→  })
   159→
   160→  const submit = () => saveMutation.mutate({ submit: true })
   161→  const saveDraft = () => saveMutation.mutate({ submit: false })
   162→
   163→  if (isLoading || !pricing) {
   164→    return (
   165→      <>
   166→        <TopBar title="Quote Builder" subtitle="Loading pricing data..." />
   167→        <div className="flex items-center justify-center py-20">
   168→          <Loader2 className="h-6 w-6 animate-spin text-brand-600" />
   169→        </div>
   170→      </>
   171→    )
   172→  }
   173→
   174→  const canSubmit = !!selection.clientId && selection.selectedFrameworkIds.length > 0 && !!selection.tierId
   175→
   176→  // Stepper gating: step 5 (Review) requires canSubmit; backward always allowed
   177→  const isStepAccessible = (stepIdx: number): boolean => {
   178→    if (stepIdx <= currentStep) return true // always allow backward
   179→    if (stepIdx === 0) return true
   180→    if (stepIdx === 5) return canSubmit // Review requires all required fields
   181→    return true // Steps 1-4 always accessible (they show "select X first" if empty)
   182→  }
   183→
   184→  return (
   185→    <>
   186→      <TopBar
   187→        title={editingQuoteId ? 'Edit Quote' : 'Quote Builder'}
   188→        subtitle={editingQuoteId ? 'Modifying an existing quote' : 'Build a compliance proposal in 6 steps'}
   189→        action={
   190→          <button
   191→            onClick={() => { reset(); toast.info(editingQuoteId ? 'Edit cancelled' : 'Quote builder reset') }}
   192→            className="inline-flex items-center gap-2 rounded-lg border border-slate-200 px-3 py-2 text-sm font-medium text-slate-600 hover:bg-slate-50"
   193→          >
   194→            <RotateCcw className="h-4 w-4" /> {editingQuoteId ? 'Cancel Edit' : 'Reset'}
   195→          </button>
   196→        }
   197→      />
   198→
   199→      <div className="flex flex-col lg:flex-row min-h-[calc(100vh-4rem)]">
   200→        {/* Left: wizard */}
   201→        <div className="flex-1 p-4 md:p-8 space-y-6 lg:max-w-[calc(100%-22rem)] min-w-0">
   202→          {/* Framework change warning */}
   203→          {frameworkChangedMidWizard && (
   204→            <div className="rounded-lg border border-amber-200 bg-amber-50 px-4 py-3 flex items-start gap-3">
   205→              <AlertTriangle className="h-5 w-5 text-amber-600 shrink-0 mt-0.5" />
   206→              <div>
   207→                <p className="text-sm font-medium text-amber-800">Framework selection changed</p>
   208→                <p className="text-xs text-amber-700 mt-0.5">
   209→                  Your framework selection changed — please review Auditor Fees and Add-ons to ensure they match your selected frameworks. Stale selections have been removed.
   210→                </p>
   211→              </div>
   212→              <button
   213→                onClick={() => useQuoteBuilderStore.setState({ frameworkChangedMidWizard: false, prevFrameworkIds: selection.selectedFrameworkIds })}
   214→                className="shrink-0 text-amber-400 hover:text-amber-600"
   215→              >
   216→                <X className="h-4 w-4" />
   217→              </button>
   218→            </div>
   219→          )}
   220→
   221→          {/* Stepper */}
   222→          <div className="flex items-center gap-1 overflow-x-auto pb-2">
   223→            {STEP_LABELS.map((label, i) => {
   224→              const done = i < currentStep
   225→              const active = i === currentStep
   226→              const accessible = isStepAccessible(i)
   227→              return (
   228→                <button
   229→                  key={label}
   230→                  onClick={() => accessible && handleSetStep(i)}
   231→                  disabled={!accessible}
   232→                  className={cn(
   233→                    'flex items-center gap-2 shrink-0',
   234→                    !accessible && 'cursor-not-allowed opacity-40',
   235→                  )}
   236→                  title={!accessible && i === 5 ? 'Complete Client, Framework, and Tier selection first' : undefined}
   237→                >
   238→                  <span
   239→                    className={cn(
   240→                      'flex h-7 w-7 items-center justify-center rounded-full text-xs font-bold transition-colors',
   241→                      done ? 'bg-brand-600 text-white' : active ? 'bg-brand-100 text-brand-700 ring-2 ring-brand-600' : accessible ? 'bg-slate-100 text-slate-400' : 'bg-slate-50 text-slate-300',
   242→                    )}
   243→                  >
   244→                    {done ? <Check className="h-3.5 w-3.5" /> : i + 1}
   245→                  </span>
   246→                  <span className={cn('text-xs font-medium hidden sm:inline', active ? 'text-brand-700' : 'text-slate-500')}>
   247→                    {label}
   248→                  </span>
   249→                  {i < STEP_LABELS.length - 1 && <ChevronRight className="h-3 w-3 text-slate-300 mx-1" />}
   250→                </button>
   251→              )
   252→            })}
   253→          </div>
   254→
   255→          {/* Step content */}
   256→          {currentStep === 0 && (
   257→            <StepFramework
   258→              selection={selection}
   259→              update={update}
   260→              pricing={pricing}
   261→              clients={clients || []}
   262→            />
   263→          )}
   264→          {currentStep === 1 && (
   265→            <StepTier selection={selection} update={update} pricing={pricing} />
   266→          )}
   267→          {currentStep === 2 && (
   268→            <StepAuditor selection={selection} update={update} pricing={pricing} />
   269→          )}
   270→          {currentStep === 3 && (
   271→            <StepAddons selection={selection} update={update} pricing={pricing} />
   272→          )}
   273→          {currentStep === 4 && (
   274→            <StepTools selection={selection} update={update} pricing={pricing} />
   275→          )}
   276→          {currentStep === 5 && (
   277→            <StepReview
   278→              selection={selection}
   279→              update={update}
   280→              pricing={pricing}
   281→              calc={calc}
   282→              clients={clients || []}
   283→              isEditing={!!editingQuoteId}
   284→            />
   285→          )}
   286→
   287→          {/* Navigation buttons */}
   288→          <div className="flex items-center justify-between pt-4 border-t border-slate-200">
   289→            <button
   290→              onClick={prevStep}
   291→              disabled={currentStep === 0}
   292→              className="inline-flex items-center gap-1.5 rounded-lg border border-slate-200 px-4 py-2 text-sm font-medium text-slate-600 hover:bg-slate-50 disabled:opacity-40 disabled:cursor-not-allowed"
   293→            >
   294→              <ChevronLeft className="h-4 w-4" /> Back
   295→            </button>
   296→
   297→            <div className="flex items-center gap-2">
   298→              {currentStep === 5 ? (
   299→                <>
   300→                  <button
   301→                    onClick={saveDraft}
   302→                    disabled={!canSubmit || saveMutation.isPending}
   303→                    className="inline-flex items-center gap-1.5 rounded-lg border border-slate-200 px-4 py-2 text-sm font-medium text-slate-700 hover:bg-slate-50 disabled:opacity-40"
   304→                  >
   305→                    <Save className="h-4 w-4" /> {editingQuoteId ? 'Update Draft' : 'Save Draft'}
   306→                  </button>
   307→                  <button
   308→                    onClick={submit}
   309→                    disabled={!canSubmit || saveMutation.isPending}
   310→                    className="inline-flex items-center gap-1.5 rounded-lg bg-brand-700 px-4 py-2 text-sm font-semibold text-white hover:bg-brand-800 disabled:opacity-40"
   311→                  >
   312→                    {saveMutation.isPending ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
   313→                    {editingQuoteId ? 'Update & Submit' : 'Submit for Approval'}
   314→                  </button>
   315→                </>
   316→              ) : (
   317→                <button
   318→                  onClick={nextStep}
   319→                  disabled={currentStep === 5}
   320→                  className="inline-flex items-center gap-1.5 rounded-lg bg-brand-700 px-4 py-2 text-sm font-semibold text-white hover:bg-brand-800"
   321→                >
   322→                  Next <ChevronRight className="h-4 w-4" />
   323→                </button>
   324→              )}
   325→            </div>
   326→          </div>
   327→        </div>
   328→
   329→        {/* Right: live summary panel (sticky on desktop) */}
   330→        <aside className="lg:w-[22rem] lg:border-l border-slate-200 bg-slate-50 p-4 md:p-6 lg:sticky lg:top-16 lg:self-start lg:max-h-[calc(100vh-4rem)] lg:overflow-y-auto">
   331→          <h3 className="text-sm font-semibold text-slate-900 mb-3 flex items-center gap-2">
   332→            <Banknote className="h-4 w-4 text-brand-700" /> Live Quote Summary
   333→          </h3>
   334→          {editingQuoteId && (
   335→            <div className="mb-3 rounded-lg bg-brand-50 border border-brand-200 px-3 py-2">
   336→              <p className="text-xs font-medium text-brand-700">
   337→                <Pencil className="inline h-3.5 w-3.5 mr-1" />
   338→                Editing existing quote
   339→              </p>
   340→            </div>
   341→          )}
   342→          <SummaryPanel selection={selection} pricing={pricing} calc={calc} />
   343→        </aside>
   344→      </div>
   345→    </>
   346→  )
   347→}

   349→// ─── Summary Panel ─────────────────────────────────────────────
   350→function SummaryPanel({ selection, pricing, calc }: {
   351→  selection: QuoteBuilderSelection
   352→  pricing: PricingData
   353→  calc: ReturnType<typeof calculateQuote> | null
   354→}) {
   355→  if (!calc) return null
   356→  const selectedFrameworks = pricing.frameworks.filter((f) => selection.selectedFrameworkIds.includes(f.id))
   357→  const tier = pricing.tiers.find((t) => t.id === selection.tierId)
   358→
   359→  return (
   360→    <div className="space-y-3">
   361→      {calc.lines.filter((l) => l.lineType !== 'internal_time').length === 0 && (
   362→        <p className="text-xs text-slate-400 py-4 text-center">
   363→          Select framework(s) and a tier to see pricing
   364→        </p>
   365→      )}
   366→      {selectedFrameworks.length > 0 && (
   367→        <p className="text-xs font-medium text-brand-700">
   368→          {selectedFrameworks.length} framework{selectedFrameworks.length > 1 ? 's' : ''} selected
   369→        </p>
   370→      )}
   371→
   372→      {calc.lines.map((line, i) => {
   373→        const isInternal = line.lineType === 'internal_time'
   374→        const isDiscount = line.lineType === 'discount'
   375→        return (
   376→          <div
   377→            key={i}
   378→            className={cn(
   379→              'flex items-start justify-between gap-2 text-sm',
   380→              isInternal && 'opacity-60',
   381→            )}
   382→          >
   383→            <span className={cn('text-slate-600 flex-1', isInternal && 'italic text-xs', isDiscount && 'text-rose-600')}>
   384→              {line.description}
   385→            </span>
   386→            <span className={cn(
   387→              'font-medium tabular-nums shrink-0',
   388→              isInternal ? 'text-slate-400 text-xs' : isDiscount ? 'text-rose-600' : 'text-slate-900',
   389→            )}>
   390→              {line.amountInr < 0 ? '-' : ''}{formatINR(Math.abs(line.amountInr))}
   391→            </span>
   392→          </div>
   393→        )
   394→      })}
   395→
   396→      <div className="border-t border-slate-200 pt-3 space-y-1.5">
   397→        <div className="flex justify-between text-sm">
   398→          <span className="text-slate-600">Subtotal</span>
   399→          <span className="font-medium text-slate-900">{formatINR(calc.subtotalInr)}</span>
   400→        </div>
   401→        {calc.discountInr > 0 && (
   402→          <div className="flex justify-between text-sm">
   403→            <span className="text-rose-600">Discount ({calc.discountPct}%)</span>
   404→            <span className="font-medium text-rose-600">-{formatINR(calc.discountInr)}</span>
   405→          </div>
   406→        )}
   407→        <div className="flex justify-between text-sm">
   408→          <span className="text-slate-600">GST @ {calc.gstRate}%</span>
   409→          <span className="font-medium text-slate-900">{formatINR(calc.gstAmountInr)}</span>
   410→        </div>
   411→        <div className="flex justify-between items-baseline pt-2 border-t border-slate-200">
   412→          <span className="text-sm font-bold text-slate-900">Grand Total</span>
   413→          <div className="text-right">
   414→            <p className="text-lg font-bold text-brand-700">{formatINR(calc.totalInr)}</p>
   415→            <p className="text-xs text-slate-500">{formatUSD(calc.totalUsd)} @ ₹{calc.usdInrRate}</p>
   416→          </div>
   417→        </div>
   418→        {calc.internalTimeCostInr > 0 && (
   419→          <p className="text-xs text-slate-400 italic pt-1">
   420→            Advisory (not billed): {formatINR(calc.internalTimeCostInr)}
   421→          </p>
   422→        )}
   423→      </div>
   424→    </div>
   425→  )
   426→}

   428→// ─── Step 0: Client + Framework ────────────────────────────────
   429→function FxRateCard({ currentRate }: { currentRate: number }) {
   430→  const queryClient = useQueryClient()
   431→  const [editing, setEditing] = useState(false)
   432→  const [draft, setDraft] = useState(String(currentRate))
   433→
   434→  const updateFx = useMutation({
   435→    mutationFn: (rate: number) => api.put('/admin/fx-rate', { rate }),
   436→    onSuccess: () => {
   437→      queryClient.invalidateQueries({ queryKey: ['pricing'] })
   438→      toast.success('FX rate updated')
   439→      setEditing(false)
   440→    },
   441→    onError: (e: Error) => toast.error(e.message),
   442→  })
   443→
   444→  const handleSave = () => {
   445→    const val = parseFloat(draft)
   446→    if (isNaN(val) || val <= 0) {
   447→      toast.error('Enter a valid positive rate')
   448→      return
   449→    }
   450→    updateFx.mutate(val)
   451→  }
   452→
   453→  const handleCancel = () => {
   454→    setDraft(String(currentRate))
   455→    setEditing(false)
   456→  }
   457→
   458→  return (
   459→    <div className="rounded-xl border border-slate-200 bg-white p-5">
   460→      <div className="flex items-start justify-between gap-4">
   461→        <div className="space-y-1">
   462→          <h3 className="text-sm font-semibold text-slate-900">USD/INR Exchange Rate</h3>
   463→          {!editing && (
   464→            <p className="text-xs text-slate-500">This rate will be used to calculate USD equivalents</p>
   465→          )}
   466→        </div>
   467→        {!editing ? (
   468→          <div className="flex items-center gap-3">
   469→            <span className="text-2xl font-bold text-brand-700 tabular-nums">{currentRate.toFixed(2)}</span>
   470→            <button
   471→              onClick={() => { setDraft(String(currentRate)); setEditing(true) }}
   472→              className="inline-flex items-center gap-1.5 rounded-lg border border-brand-200 bg-brand-50 px-3 py-1.5 text-xs font-medium text-brand-700 hover:bg-brand-100 transition-colors"
   473→            >
   474→              <Pencil className="h-3 w-3" /> Update Rate
   475→            </button>
   476→          </div>
   477→        ) : (
   478→          <div className="flex items-center gap-2">
   479→            <input
   480→              type="number"
   481→              step={0.01}
   482→              min={0.01}
   483→              value={draft}
   484→              onChange={(e) => setDraft(e.target.value)}
   485→              onKeyDown={(e) => { if (e.key === 'Enter') handleSave(); if (e.key === 'Escape') handleCancel() }}
   486→              className="w-28 rounded-lg border border-slate-200 px-2.5 py-1.5 text-sm tabular-nums focus:outline-none focus:ring-2 focus:ring-brand-600"
   487→              autoFocus
   488→            />
   489→            <button
   490→              onClick={handleSave}
   491→              disabled={updateFx.isPending}
   492→              className="inline-flex items-center gap-1 rounded-lg bg-brand-700 px-3 py-1.5 text-xs font-semibold text-white hover:bg-brand-800 disabled:opacity-50 transition-colors"
   493→            >
   494→              {updateFx.isPending ? <Loader2 className="h-3 w-3 animate-spin" /> : <Check className="h-3 w-3" />} Save
   495→            </button>
   496→            <button
   497→              onClick={handleCancel}
   498→              disabled={updateFx.isPending}
   499→              className="inline-flex items-center gap-1 rounded-lg border border-slate-200 px-3 py-1.5 text-xs font-medium text-slate-600 hover:bg-slate-50 disabled:opacity-50 transition-colors"
   500→            >
   501→              Cancel
   502→            </button>
   503→          </div>
   504→        )}
   505→      </div>
   506→    </div>
   507→  )
   508→}

   510→function StepFramework({ selection, update, pricing, clients }: {
   511→  selection: QuoteBuilderSelection
   512→  update: (p: Partial<QuoteBuilderSelection>) => void
   513→  pricing: PricingData
