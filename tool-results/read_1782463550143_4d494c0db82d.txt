   252→   252→function SummaryPanel({ selection, pricing, calc }: {
   253→   253→  selection: QuoteBuilderSelection

   304→   304→          <div className="flex justify-between text-sm">

   331→   331→function FxRateCard({ currentRate }: { currentRate: number }) {
   332→   332→  const queryClient = useQueryClient()
   333→   333→  const [editing, setEditing] = useState(false)
   334→   334→  const [draft, setDraft] = useState(String(currentRate))
   335→   335→
   336→   336→  const updateFx = useMutation({
   337→   337→    mutationFn: (rate: number) => api.put('/admin/fx-rate', { rate }),
   338→   338→    onSuccess: () => {
   339→   339→      queryClient.invalidateQueries({ queryKey: ['pricing'] })
   340→   340→      toast.success('FX rate updated')
   341→   341→      setEditing(false)
   342→   342→    },
   343→   343→    onError: (e: Error) => toast.error(e.message),
   344→   344→  })
   345→   345→
   346→   346→  const handleSave = () => {
   347→   347→    const val = parseFloat(draft)
   348→   348→    if (isNaN(val) || val <= 0) {
   349→   349→      toast.error('Enter a valid positive rate')
   350→   350→      return
   351→   351→    }
   352→   352→    updateFx.mutate(val)
   353→   353→  }
   354→   354→
   355→   355→  const handleCancel = () => {
   356→   356→    setDraft(String(currentRate))
   357→   357→    setEditing(false)
   358→   358→  }
   359→   359→
   360→   360→  return (
   361→   361→    <div className="rounded-xl border border-slate-200 bg-white p-5">
   362→   362→      <div className="flex items-start justify-between gap-4">
   363→   363→        <div className="space-y-1">
   364→   364→          <h3 className="text-sm font-semibold text-slate-900">USD/INR Exchange Rate</h3>
   365→   365→          {!editing && (
   366→   366→            <p className="text-xs text-slate-500">This rate will be used to calculate USD equivalents</p>
   367→   367→          )}
   368→   368→        </div>
   369→   369→        {!editing ? (
   370→   370→          <div className="flex items-center gap-3">
   371→   371→            <span className="text-2xl font-bold text-brand-700 tabular-nums">{currentRate.toFixed(2)}</span>
   372→   372→            <button
   373→   373→              onClick={() => { setDraft(String(currentRate)); setEditing(true) }}
   374→   374→              className="inline-flex items-center gap-1.5 rounded-lg border border-brand-200 bg-brand-50 px-3 py-1.5 text-xs font-medium text-brand-700 hover:bg-brand-100 transition-colors"
   375→   375→            >
   376→   376→              <Pencil className="h-3 w-3" /> Update Rate
   377→   377→            </button>
   378→   378→          </div>
   379→   379→        ) : (
   380→   380→          <div className="flex items-center gap-2">
   381→   381→            <input
   382→   382→              type="number"
   383→   383→              step={0.01}
   384→   384→              min={0.01}
   385→   385→              value={draft}
   386→   386→              onChange={(e) => setDraft(e.target.value)}
   387→   387→              onKeyDown={(e) => { if (e.key === 'Enter') handleSave(); if (e.key === 'Escape') handleCancel() }}
   388→   388→              className="w-28 rounded-lg border border-slate-200 px-2.5 py-1.5 text-sm tabular-nums focus:outline-none focus:ring-2 focus:ring-brand-600"
   389→   389→              autoFocus
   390→   390→            />
   391→   391→            <button
   392→   392→              onClick={handleSave}
   393→   393→              disabled={updateFx.isPending}
   394→   394→              className="inline-flex items-center gap-1 rounded-lg bg-brand-700 px-3 py-1.5 text-xs font-semibold text-white hover:bg-brand-800 disabled:opacity-50 transition-colors"
   395→   395→            >
   396→   396→              {updateFx.isPending ? <Loader2 className="h-3 w-3 animate-spin" /> : <Check className="h-3 w-3" />} Save
   397→   397→            </button>
   398→   398→            <button
   399→   399→              onClick={handleCancel}
   400→   400→              disabled={updateFx.isPending}
   401→   401→              className="inline-flex items-center gap-1 rounded-lg border border-slate-200 px-3 py-1.5 text-xs font-medium text-slate-600 hover:bg-slate-50 disabled:opacity-50 transition-colors"
   402→   402→            >
   403→   403→              Cancel
   404→   404→            </button>
   405→   405→          </div>
   406→   406→        )}
   407→   407→      </div>
   408→   408→    </div>
   409→   409→  )
   410→   410→}

   412→   412→function StepFramework({ selection, update, pricing, clients }: {
   413→   413→  selection: QuoteBuilderSelection
   414→   414→  update: (p: Partial<QuoteBuilderSelection>) => void
   415→   415→  pricing: PricingData
   416→   416→  clients: Client[]
   417→   417→}) {
   418→   418→  const grouped = pricing.frameworks.reduce((acc, fw) => {
   419→   419→    (acc[fw.category] ||= []).push(fw)
   420→   420→    return acc
   421→   421→  }, {} as Record<string, Framework[]>)
   422→   422→
   423→   423→  const toggle = (id: string) => {
   424→   424→    const ids = selection.selectedFrameworkIds
   425→   425→    update({
   426→   426→      selectedFrameworkIds: ids.includes(id)
   427→   427→        ? ids.filter((x) => x !== id)
   428→   428→        : [...ids, id],
   429→   429→    })
   430→   430→  }
   431→   431→
   432→   432→  const clearAll = () => update({ selectedFrameworkIds: [] })
   433→   433→  const selectAll = () => update({ selectedFrameworkIds: pricing.frameworks.map((f) => f.id) })
   434→   434→
   435→   435→  return (
   436→   436→    <div className="space-y-6">
   437→   437→      {/* FX Rate */}
   438→   438→      <FxRateCard currentRate={pricing.fxRate} />
   439→   439→
   440→   440→      {/* Client selection */}
   441→   441→      <SectionCard title="1. Select Client" description="Choose the prospect this quote is for">
