   252→   252→function SummaryPanel({ selection, pricing, calc }: {

   299→   299→          <div className="flex justify-between text-sm">

   303→   303→        )}

   326→   326→function FxRateCard({ currentRate }: { currentRate: number }) {
   327→   327→  const queryClient = useQueryClient()
   328→   328→  const [editing, setEditing] = useState(false)
   329→   329→  const [draft, setDraft] = useState(String(currentRate))
   330→   330→
   331→   331→  const updateFx = useMutation({
   332→   332→    mutationFn: (rate: number) => api.put('/admin/fx-rate', { rate }),
   333→   333→    onSuccess: () => {
   334→   334→      queryClient.invalidateQueries({ queryKey: ['pricing'] })
   335→   335→      toast.success('FX rate updated')
   336→   336→      setEditing(false)
   337→   337→    },
   338→   338→    onError: (e: Error) => toast.error(e.message),
   339→   339→  })
   340→   340→
   341→   341→  const handleSave = () => {
   342→   342→    const val = parseFloat(draft)
   343→   343→    if (isNaN(val) || val <= 0) {
   344→   344→      toast.error('Enter a valid positive rate')
   345→   345→      return
   346→   346→    }
   347→   347→    updateFx.mutate(val)
   348→   348→  }
   349→   349→
   350→   350→  const handleCancel = () => {
   351→   351→    setDraft(String(currentRate))
   352→   352→    setEditing(false)
   353→   353→  }
   354→   354→
   355→   355→  return (
   356→   356→    <div className="rounded-xl border border-slate-200 bg-white p-5">
   357→   357→      <div className="flex items-start justify-between gap-4">
   358→   358→        <div className="space-y-1">
   359→   359→          <h3 className="text-sm font-semibold text-slate-900">USD/INR Exchange Rate</h3>
   360→   360→          {!editing && (
   361→   361→            <p className="text-xs text-slate-500">This rate will be used to calculate USD equivalents</p>
   362→   362→          )}
   363→   363→        </div>
   364→   364→        {!editing ? (
   365→   365→          <div className="flex items-center gap-3">
   366→   366→            <span className="text-2xl font-bold text-brand-700 tabular-nums">{currentRate.toFixed(2)}</span>
   367→   367→            <button
   368→   368→              onClick={() => { setDraft(String(currentRate)); setEditing(true) }}
   369→   369→              className="inline-flex items-center gap-1.5 rounded-lg border border-brand-200 bg-brand-50 px-3 py-1.5 text-xs font-medium text-brand-700 hover:bg-brand-100 transition-colors"
   370→   370→            >
   371→   371→              <Pencil className="h-3 w-3" /> Update Rate
   372→   372→            </button>
   373→   373→          </div>
   374→   374→        ) : (
   375→   375→          <div className="flex items-center gap-2">
   376→   376→            <input
   377→   377→              type="number"
   378→   378→              step={0.01}
   379→   379→              min={0.01}
   380→   380→              value={draft}
   381→   381→              onChange={(e) => setDraft(e.target.value)}
   382→   382→              onKeyDown={(e) => { if (e.key === 'Enter') handleSave(); if (e.key === 'Escape') handleCancel() }}
   383→   383→              className="w-28 rounded-lg border border-slate-200 px-2.5 py-1.5 text-sm tabular-nums focus:outline-none focus:ring-2 focus:ring-brand-600"
   384→   384→              autoFocus
   385→   385→            />
   386→   386→            <button
   387→   387→              onClick={handleSave}
   388→   388→              disabled={updateFx.isPending}
   389→   389→              className="inline-flex items-center gap-1 rounded-lg bg-brand-700 px-3 py-1.5 text-xs font-semibold text-white hover:bg-brand-800 disabled:opacity-50 transition-colors"
   390→   390→            >
   391→   391→              {updateFx.isPending ? <Loader2 className="h-3 w-3 animate-spin" /> : <Check className="h-3 w-3" />} Save
   392→   392→            </button>
   393→   393→            <button
   394→   394→              onClick={handleCancel}
   395→   395→              disabled={updateFx.isPending}
   396→   396→              className="inline-flex items-center gap-1 rounded-lg border border-slate-200 px-3 py-1.5 text-xs font-medium text-slate-600 hover:bg-slate-50 disabled:opacity-50 transition-colors"
   397→   397→            >
   398→   398→              Cancel
   399→   399→            </button>
   400→   400→          </div>
   401→   401→        )}
   402→   402→      </div>
   403→   403→    </div>
   404→   404→  )
   405→   405→}

   407→   407→function StepFramework({ selection, update, pricing, clients }: {
   408→   408→  selection: QuoteBuilderSelection
   409→   409→  update: (p: Partial<QuoteBuilderSelection>) => void
   410→   410→  pricing: PricingData
   411→   411→  clients: Client[]
   412→   412→}) {
   413→   413→  const grouped = pricing.frameworks.reduce((acc, fw) => {
   414→   414→    (acc[fw.category] ||= []).push(fw)
   415→   415→    return acc
   416→   416→  }, {} as Record<string, Framework[]>)
   417→   417→
   418→   418→  return (
   419→   419→    <div className="space-y-6">
   420→   420→      {/* FX Rate */}
   421→   421→      <FxRateCard currentRate={pricing.fxRate} />
   422→   422→
   423→   423→      {/* Client selection */}
   424→   424→      <SectionCard title="1. Select Client" description="Choose the prospect this quote is for">
   425→   425→        <select
   426→   426→          value={selection.clientId || ''}
   427→   427→          onChange={(e) => update({ clientId: e.target.value || null })}
   428→   428→          className="w-full rounded-lg border border-slate-200 px-3 py-2.5 text-sm focus:outline-none focus:ring-2 focus:ring-brand-600"
   429→   429→        >
   430→   430→          <option value="">— Select a client —</option>
   431→   431→          {clients.map((c) => (
   432→   432→            <option key={c.id} value={c.id}>
   433→   433→              {c.companyName} ({c.sector || 'N/A'} • {c.companySize})
   434→   434→            </option>
   435→   435→          ))}
   436→   436→        </select>
   437→   437→      </SectionCard>
   438→   438→
   439→   439→      {/* Framework selection */}
