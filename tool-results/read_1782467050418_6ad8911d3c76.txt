   525→// ─── Animated Counter ──────────────────────────────────────────
   526→function AnimatedCounter({ value, duration = 1200 }: { value: number; duration?: number }) {
   527→  const [display, setDisplay] = useState(0)
   528→  const prevValue = useRef(0)
   529→
   530→  useEffect(() => {
   531→    const start = prevValue.current
   532→    const end = value
   533→    const startTime = performance.now()
   534→
   535→    const animate = (currentTime: number) => {
   536→      const elapsed = currentTime - startTime
   537→      const progress = Math.min(elapsed / duration, 1)
   538→      // Ease out cubic
   539→      const eased = 1 - Math.pow(1 - progress, 3)
   540→      setDisplay(Math.round(start + (end - start) * eased))
   541→      if (progress < 1) requestAnimationFrame(animate)
   542→    }
   543→
   544→    requestAnimationFrame(animate)
   545→    prevValue.current = value
   546→  }, [value, duration])
   547→
   548→  return <>{formatNumberIN(display)}</>
   549→}

   551→// ─── Export CSV ────────────────────────────────────────────────
   552→function exportCsv() {
   553→  const rows = [
   554→    ['Client', 'Framework', 'Tier', 'Status', 'Subtotal (INR)', 'Discount (INR)', 'GST (INR)', 'Total (INR)', 'USD', 'Created'],
   555→  ]
   556→
   557→  // We'll fetch fresh data for export
   558→  fetch('/api/quotes?status=all')
   559→    .then((r) => r.json())
   560→    .then((res) => {
   561→      const quotes = res.data || []
   562→      for (const q of quotes) {
   563→        rows.push([
   564→          q.client?.companyName || '',
   565→          q.framework?.name || '',
   566→          q.tier?.name || '',
   567→          q.status,
   568→          String(q.subtotalInr),
   569→          String(q.discountInr),
   570→          String(q.gstAmountInr),
   571→          String(q.totalInr),
   572→          String(q.totalUsd),
   573→          new Date(q.createdAt).toLocaleDateString('en-IN'),
   574→        ])
   575→      }
   576→      const csv = rows.map((r) => r.map((c) => `"${c}"`).join(',')).join('\n')
   577→      const blob = new Blob([csv], { type: 'text/csv' })
   578→      const url = URL.createObjectURL(blob)
   579→      const a = document.createElement('a')
   580→      a.href = url
   581→      a.download = `isecurify-quotes-${new Date().toISOString().split('T')[0]}.csv`
   582→      document.body.appendChild(a)
   583→      a.click()
   584→      document.body.removeChild(a)
   585→      URL.revokeObjectURL(url)
   586→    })
   587→    .catch(() => {
   588→      // fallback: ignore
   589→    })
   590→}
