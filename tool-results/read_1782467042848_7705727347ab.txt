   525→   525→   525→// ─── Animated Counter ──────────────────────────────────────────
   526→   526→   526→function AnimatedCounter({ value, duration = 1200 }: { value: number; duration?: number }) {
   527→   527→   527→  const [display, setDisplay] = useState(0)
   528→   528→   528→  const prevValue = useRef(0)
   529→   529→   529→
   530→   530→   530→  useEffect(() => {
   531→   531→   531→    const start = prevValue.current
   532→   532→   532→    const end = value
   533→   533→   533→    const startTime = performance.now()
   534→   534→   534→
   535→   535→   535→    const animate = (currentTime: number) => {
   536→   536→   536→      const elapsed = currentTime - startTime
   537→   537→   537→      const progress = Math.min(elapsed / duration, 1)
   538→   538→   538→      // Ease out cubic
   539→   539→   539→      const eased = 1 - Math.pow(1 - progress, 3)
   540→   540→   540→      setDisplay(Math.round(start + (end - start) * eased))
   541→   541→   541→      if (progress < 1) requestAnimationFrame(animate)
   542→   542→   542→    }
   543→   543→   543→
   544→   544→   544→    requestAnimationFrame(animate)
   545→   545→   545→    prevValue.current = value
   546→   546→   546→  }, [value, duration])
   547→   547→   547→
   548→   548→   548→  return <>{formatNumberIN(display)}</>
   549→   549→   549→}

   552→   552→   552→function exportCsv() {
   553→   553→   553→  const rows = [
   554→   554→   554→    ['Client', 'Framework', 'Tier', 'Status', 'Subtotal (INR)', 'Discount (INR)', 'GST (INR)', 'Total (INR)', 'USD', 'Created'],
   555→   555→   555→  ]
   556→   556→   556→
   557→   557→   557→  // We'll fetch fresh data for export
   558→   558→   558→  fetch('/api/quotes?status=all')
   559→   559→   559→    .then((r) => r.json())
   560→   560→   560→    .then((res) => {
   561→   561→   561→      const quotes = res.data || []
   562→   562→   562→      for (const q of quotes) {
   563→   563→   563→        rows.push([
   564→   564→   564→          q.client?.companyName || '',
   565→   565→   565→          q.framework?.name || '',
   566→   566→   566→          q.tier?.name || '',
   567→   567→   567→          q.status,
   568→   568→   568→          String(q.subtotalInr),
   569→   569→   569→          String(q.discountInr),
   570→   570→   570→          String(q.gstAmountInr),
   571→   571→   571→          String(q.totalInr),
   572→   572→   572→          String(q.totalUsd),
   573→   573→   573→          new Date(q.createdAt).toLocaleDateString('en-IN'),
   574→   574→   574→        ])
   575→   575→   575→      }
   576→   576→   576→      const csv = rows.map((r) => r.map((c) => `"${c}"`).join(',')).join('\n')
   577→   577→   577→      const blob = new Blob([csv], { type: 'text/csv' })
   578→   578→   578→      const url = URL.createObjectURL(blob)
   579→   579→   579→      const a = document.createElement('a')
   580→   580→   580→      a.href = url
   581→   581→   581→      a.download = `isecurify-quotes-${new Date().toISOString().split('T')[0]}.csv`
   582→   582→   582→      document.body.appendChild(a)
   583→   583→   583→      a.click()
   584→   584→   584→      document.body.removeChild(a)
   585→   585→   585→      URL.revokeObjectURL(url)
   586→   586→   586→    })
   587→   587→   587→    .catch(() => {
   588→   588→   588→      // fallback: ignore
   589→   589→   589→    })
   590→   590→   590→}
