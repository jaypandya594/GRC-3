   251→// ─── Summary Panel ─────────────────────────────────────────────
   252→function SummaryPanel({ selection, pricing, calc }: {
   253→  selection: QuoteBuilderSelection
   254→  pricing: PricingData
   255→  calc: ReturnType<typeof calculateQuote> | null
   256→}) {
   257→  if (!calc) return null
   258→  const selectedFrameworks = pricing.frameworks.filter((f) => selection.selectedFrameworkIds.includes(f.id))
   259→  const tier = pricing.tiers.find((t) => t.id === selection.tierId)
   260→
   261→  return (
   262→    <div className="space-y-3">
   263→      {calc.lines.filter((l) => l.lineType !== 'internal_time').length === 0 && (
   264→        <p className="text-xs text-slate-400 py-4 text-center">
   265→          Select framework(s) and a tier to see pricing
   266→        </p>
   267→      )}
   268→      {selectedFrameworks.length > 0 && (
   269→        <p className="text-xs font-medium text-brand-700">
   270→          {selectedFrameworks.length} framework{selectedFrameworks.length > 1 ? 's' : ''} selected
   271→        </p>
   272→      )}
   273→
   274→      {calc.lines.map((line, i) => {
   275→        const isInternal = line.lineType === 'internal_time'
   276→        const isDiscount = line.lineType === 'discount'
   277→        return (
   278→          <div
   279→            key={i}
   280→            className={cn(
   281→              'flex items-start justify-between gap-2 text-sm',
   282→              isInternal && 'opacity-60',
   283→            )}
   284→          >
   285→            <span className={cn('text-slate-600 flex-1', isInternal && 'italic text-xs', isDiscount && 'text-rose-600')}>
   286→              {line.description}
   287→            </span>
   288→            <span className={cn(
   289→              'font-medium tabular-nums shrink-0',
   290→              isInternal ? 'text-slate-400 text-xs' : isDiscount ? 'text-rose-600' : 'text-slate-900',
   291→            )}>
   292→              {line.amountInr < 0 ? '-' : ''}{formatINR(Math.abs(line.amountInr))}
   293→            </span>
   294→          </div>
   295→        )
   296→      })}
   297→
   298→      <div className="border-t border-slate-200 pt-3 space-y-1.5">
   299→        <div className="flex justify-between text-sm">
   300→          <span className="text-slate-600">Subtotal</span>
   301→          <span className="font-medium text-slate-900">{formatINR(calc.subtotalInr)}</span>
   302→        </div>
   303→        {calc.discountInr > 0 && (
   304→          <div className="flex justify-between text-sm">
   305→            <span className="text-rose-600">Discount ({calc.discountPct}%)</span>
   306→            <span className="font-medium text-rose-600">-{formatINR(calc.discountInr)}</span>
   307→          </div>
   308→        )}
   309→        <div className="flex justify-between text-sm">
   310→          <span className="text-slate-600">GST @ {calc.gstRate}%</span>
   311→          <span className="font-medium text-slate-900">{formatINR(calc.gstAmountInr)}</span>
   312→        </div>
   313→        <div className="flex justify-between items-baseline pt-2 border-t border-slate-200">
   314→          <span className="text-sm font-bold text-slate-900">Grand Total</span>
   315→          <div className="text-right">
   316→            <p className="text-lg font-bold text-brand-700">{formatINR(calc.totalInr)}</p>
   317→            <p className="text-xs text-slate-500">{formatUSD(calc.totalUsd)} @ ₹{calc.usdInrRate}</p>
   318→          </div>
   319→        </div>
   320→        {calc.internalTimeCostInr > 0 && (
   321→          <p className="text-xs text-slate-400 italic pt-1">
   322→            Advisory (not billed): {formatINR(calc.internalTimeCostInr)}
   323→          </p>
   324→        )}
   325→      </div>
   326→    </div>
   327→  )
   328→}

   330→// ─── Step 0: Client + Framework ────────────────────────────────
   331→function FxRateCard({ currentRate }: { currentRate: number }) {
   332→  const queryClient = useQueryClient()
   333→  const [editing, setEditing] = useState(false)
   334→  const [draft, setDraft] = useState(String(currentRate))
   335→
   336→  const updateFx = useMutation({
   337→    mutationFn: (rate: number) => api.put('/admin/fx-rate', { rate }),
   338→    onSuccess: () => {
   339→      queryClient.invalidateQueries({ queryKey: ['pricing'] })
   340→      toast.success('FX rate updated')
   341→      setEditing(false)
   342→    },
   343→    onError: (e: Error) => toast.error(e.message),
   344→  })
   345→
   346→  const handleSave = () => {
   347→    const val = parseFloat(draft)
   348→    if (isNaN(val) || val <= 0) {
   349→      toast.error('Enter a valid positive rate')
   350→      return
   351→    }
   352→    updateFx.mutate(val)
   353→  }
   354→
   355→  const handleCancel = () => {
   356→    setDraft(String(currentRate))
   357→    setEditing(false)
   358→  }
   359→
   360→  return (
   361→    <div className="rounded-xl border border-slate-200 bg-white p-5">
   362→      <div className="flex items-start justify-between gap-4">
   363→        <div className="space-y-1">
   364→          <h3 className="text-sm font-semibold text-slate-900">USD/INR Exchange Rate</h3>
   365→          {!editing && (
   366→            <p className="text-xs text-slate-500">This rate will be used to calculate USD equivalents</p>
   367→          )}
   368→        </div>
   369→        {!editing ? (
   370→          <div className="flex items-center gap-3">
   371→            <span className="text-2xl font-bold text-brand-700 tabular-nums">{currentRate.toFixed(2)}</span>
   372→            <button
   373→              onClick={() => { setDraft(String(currentRate)); setEditing(true) }}
   374→              className="inline-flex items-center gap-1.5 rounded-lg border border-brand-200 bg-brand-50 px-3 py-1.5 text-xs font-medium text-brand-700 hover:bg-brand-100 transition-colors"
   375→            >
   376→              <Pencil className="h-3 w-3" /> Update Rate
   377→            </button>
   378→          </div>
   379→        ) : (
   380→          <div className="flex items-center gap-2">
   381→            <input
   382→              type="number"
   383→              step={0.01}
   384→              min={0.01}
   385→              value={draft}
   386→              onChange={(e) => setDraft(e.target.value)}
   387→              onKeyDown={(e) => { if (e.key === 'Enter') handleSave(); if (e.key === 'Escape') handleCancel() }}
   388→              className="w-28 rounded-lg border border-slate-200 px-2.5 py-1.5 text-sm tabular-nums focus:outline-none focus:ring-2 focus:ring-brand-600"
   389→              autoFocus
   390→            />
   391→            <button
   392→              onClick={handleSave}
   393→              disabled={updateFx.isPending}
   394→              className="inline-flex items-center gap-1 rounded-lg bg-brand-700 px-3 py-1.5 text-xs font-semibold text-white hover:bg-brand-800 disabled:opacity-50 transition-colors"
   395→            >
   396→              {updateFx.isPending ? <Loader2 className="h-3 w-3 animate-spin" /> : <Check className="h-3 w-3" />} Save
   397→            </button>
   398→            <button
   399→              onClick={handleCancel}
   400→              disabled={updateFx.isPending}
   401→              className="inline-flex items-center gap-1 rounded-lg border border-slate-200 px-3 py-1.5 text-xs font-medium text-slate-600 hover:bg-slate-50 disabled:opacity-50 transition-colors"
   402→            >
   403→              Cancel
   404→            </button>
   405→          </div>
   406→        )}
   407→      </div>
   408→    </div>
   409→  )
   410→}

   412→function StepFramework({ selection, update, pricing, clients }: {
   413→  selection: QuoteBuilderSelection
   414→  update: (p: Partial<QuoteBuilderSelection>) => void
   415→  pricing: PricingData
   416→  clients: Client[]
   417→}) {
   418→  const grouped = pricing.frameworks.reduce((acc, fw) => {
   419→    (acc[fw.category] ||= []).push(fw)
   420→    return acc
   421→  }, {} as Record<string, Framework[]>)
   422→
   423→  const toggle = (id: string) => {
   424→    const ids = selection.selectedFrameworkIds
   425→    update({
   426→      selectedFrameworkIds: ids.includes(id)
   427→        ? ids.filter((x) => x !== id)
   428→        : [...ids, id],
   429→    })
   430→  }
   431→
   432→  const clearAll = () => update({ selectedFrameworkIds: [] })
   433→  const selectAll = () => update({ selectedFrameworkIds: pricing.frameworks.map((f) => f.id) })
   434→
   435→  return (
   436→    <div className="space-y-6">
   437→      {/* FX Rate */}
   438→      <FxRateCard currentRate={pricing.fxRate} />
   439→
   440→      {/* Client selection */}
   441→      <SectionCard title="1. Select Client" description="Choose the prospect this quote is for">
   442→        <select
   443→          value={selection.clientId || ''}
   444→          onChange={(e) => update({ clientId: e.target.value || null })}
   445→          className="w-full rounded-lg border border-slate-200 px-3 py-2.5 text-sm focus:outline-none focus:ring-2 focus:ring-brand-600"
   446→        >
   447→          <option value="">— Select a client —</option>
   448→          {clients.map((c) => (
   449→            <option key={c.id} value={c.id}>
   450→              {c.companyName} ({c.sector || 'N/A'} • {c.companySize})
   451→            </option>
   452→          ))}
   453→        </select>
   454→      </SectionCard>
   455→
   456→      {/* Framework selection — MULTI */}
   457→      <SectionCard
   458→        title="2. Select Framework(s)"
   459→        description="Choose one or more compliance services — all selected frameworks use the same tier"
   460→        action={
   461→          <div className="flex items-center gap-3">
   462→            {selection.selectedFrameworkIds.length > 0 && (
   463→              <button onClick={clearAll} className="text-xs font-medium text-rose-600 hover:text-rose-700">
   464→                Clear All
   465→              </button>
   466→            )}
   467→            {selection.selectedFrameworkIds.length < pricing.frameworks.length && (
   468→              <button onClick={selectAll} className="text-xs font-medium text-brand-700 hover:text-brand-800">
   469→                Select All
   470→              </button>
   471→            )}
   472→          </div>
   473→        }
   474→      >
   475→        {selection.selectedFrameworkIds.length > 0 && (
   476→          <div className="mb-4 rounded-lg bg-brand-50 border border-brand-200 px-3 py-2">
   477→            <p className="text-xs font-medium text-brand-700">
   478→              <Layers className="inline h-3.5 w-3.5 mr-1" />
   479→              {selection.selectedFrameworkIds.length} framework{selection.selectedFrameworkIds.length > 1 ? 's' : ''} selected
   480→            </p>
   481→          </div>
   482→        )}
   483→        {Object.entries(grouped).map(([category, fws]) => {
   484→          const meta = FRAMEWORK_CATEGORY_META[category] || { label: category, color: 'slate' }
   485→          return (
   486→            <div key={category} className="mb-6 last:mb-0">
   487→              <h4 className="text-xs font-bold uppercase tracking-wide text-slate-500 mb-3">{meta.label}</h4>
   488→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">
   489→                {fws.map((fw) => (
   490→                  <PricingCard
   491→                    key={fw.id}
   492→                    selected={selection.selectedFrameworkIds.includes(fw.id)}
   493→                    onClick={() => toggle(fw.id)}
   494→                    title={fw.name}
   495→                    description={fw.description || undefined}
   496→                    badge={meta.label}
   497→                  />
   498→                ))}
   499→              </div>
   500→            </div>
   501→          )
   502→        })}
   503→      </SectionCard>
   504→    </div>
   505→  )
   506→}

   508→// ─── Step 1: Tier ──────────────────────────────────────────────
   509→function StepTier({ selection, update, pricing }: {
   510→  selection: QuoteBuilderSelection
   511→  update: (p: Partial<QuoteBuilderSelection>) => void
   512→  pricing: PricingData
   513→}) {
   514→  const selectedFws = pricing.frameworks.filter((f) => selection.selectedFrameworkIds.includes(f.id))
   515→  if (selectedFws.length === 0) {
   516→    return (
   517→      <SectionCard title="Select Tier">
   518→        <EmptyState title="No framework(s) selected" description="Go back to Step 1 and select at least one framework." />
   519→      </SectionCard>
   520→    )
   521→  }
   522→
   523→  return (
   524→    <SectionCard
   525→      title="3. Select Tier"
   526→      description={`${selectedFws.length === 1 ? selectedFws[0].name : `${selectedFws.length} frameworks`} — pricing varies by company size`}
   527→    >
   528→      <div className="grid grid-cols-1 md:grid-cols-3 gap-4 mb-6">

   630→          </div>
