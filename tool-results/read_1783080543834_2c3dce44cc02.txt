   267→   267→function SummaryPanel({ selection, pricing, calc }: {
   268→   268→  selection: QuoteBuilderSelection

   319→   319→          <div className="flex justify-between text-sm">

   346→   346→function FxRateCard({ currentRate }: { currentRate: number }) {
   347→   347→  const queryClient = useQueryClient()
   348→   348→  const [editing, setEditing] = useState(false)
   349→   349→  const [draft, setDraft] = useState(String(currentRate))
   350→   350→
   351→   351→  const updateFx = useMutation({
   352→   352→    mutationFn: (rate: number) => api.put('/admin/fx-rate', { rate }),
   353→   353→    onSuccess: () => {
   354→   354→      queryClient.invalidateQueries({ queryKey: ['pricing'] })
   355→   355→      toast.success('FX rate updated')
   356→   356→      setEditing(false)
   357→   357→    },
   358→   358→    onError: (e: Error) => toast.error(e.message),
   359→   359→  })
   360→   360→
   361→   361→  const handleSave = () => {
   362→   362→    const val = parseFloat(draft)
   363→   363→    if (isNaN(val) || val <= 0) {
   364→   364→      toast.error('Enter a valid positive rate')
   365→   365→      return
   366→   366→    }
   367→   367→    updateFx.mutate(val)
   368→   368→  }
   369→   369→
   370→   370→  const handleCancel = () => {
   371→   371→    setDraft(String(currentRate))
   372→   372→    setEditing(false)
   373→   373→  }
   374→   374→
   375→   375→  return (
   376→   376→    <div className="rounded-xl border border-slate-200 bg-white p-5">
   377→   377→      <div className="flex items-start justify-between gap-4">
   378→   378→        <div className="space-y-1">
   379→   379→          <h3 className="text-sm font-semibold text-slate-900">USD/INR Exchange Rate</h3>
   380→   380→          {!editing && (
   381→   381→            <p className="text-xs text-slate-500">This rate will be used to calculate USD equivalents</p>
   382→   382→          )}
   383→   383→        </div>
   384→   384→        {!editing ? (
   385→   385→          <div className="flex items-center gap-3">
   386→   386→            <span className="text-2xl font-bold text-brand-700 tabular-nums">{currentRate.toFixed(2)}</span>
   387→   387→            <button
   388→   388→              onClick={() => { setDraft(String(currentRate)); setEditing(true) }}
   389→   389→              className="inline-flex items-center gap-1.5 rounded-lg border border-brand-200 bg-brand-50 px-3 py-1.5 text-xs font-medium text-brand-700 hover:bg-brand-100 transition-colors"
   390→   390→            >
   391→   391→              <Pencil className="h-3 w-3" /> Update Rate
   392→   392→            </button>
   393→   393→          </div>
   394→   394→        ) : (
   395→   395→          <div className="flex items-center gap-2">
   396→   396→            <input
   397→   397→              type="number"
   398→   398→              step={0.01}
   399→   399→              min={0.01}
   400→   400→              value={draft}
   401→   401→              onChange={(e) => setDraft(e.target.value)}
   402→   402→              onKeyDown={(e) => { if (e.key === 'Enter') handleSave(); if (e.key === 'Escape') handleCancel() }}
   403→   403→              className="w-28 rounded-lg border border-slate-200 px-2.5 py-1.5 text-sm tabular-nums focus:outline-none focus:ring-2 focus:ring-brand-600"
   404→   404→              autoFocus
   405→   405→            />
   406→   406→            <button
   407→   407→              onClick={handleSave}
   408→   408→              disabled={updateFx.isPending}
   409→   409→              className="inline-flex items-center gap-1 rounded-lg bg-brand-700 px-3 py-1.5 text-xs font-semibold text-white hover:bg-brand-800 disabled:opacity-50 transition-colors"
   410→   410→            >
   411→   411→              {updateFx.isPending ? <Loader2 className="h-3 w-3 animate-spin" /> : <Check className="h-3 w-3" />} Save
   412→   412→            </button>
   413→   413→            <button
   414→   414→              onClick={handleCancel}
   415→   415→              disabled={updateFx.isPending}
   416→   416→              className="inline-flex items-center gap-1 rounded-lg border border-slate-200 px-3 py-1.5 text-xs font-medium text-slate-600 hover:bg-slate-50 disabled:opacity-50 transition-colors"
   417→   417→            >
   418→   418→              Cancel
   419→   419→            </button>
   420→   420→          </div>
   421→   421→        )}
   422→   422→      </div>
   423→   423→    </div>
   424→   424→  )
   425→   425→}

   427→   427→function StepFramework({ selection, update, pricing, clients }: {
   428→   428→  selection: QuoteBuilderSelection
   429→   429→  update: (p: Partial<QuoteBuilderSelection>) => void
   430→   430→  pricing: PricingData
   431→   431→  clients: Client[]
   432→   432→}) {
   433→   433→  const grouped = pricing.frameworks.reduce((acc, fw) => {
   434→   434→    (acc[fw.category] ||= []).push(fw)
   435→   435→    return acc
   436→   436→  }, {} as Record<string, Framework[]>)
   437→   437→
   438→   438→  const toggle = (id: string) => {
   439→   439→    const ids = selection.selectedFrameworkIds
   440→   440→    update({
