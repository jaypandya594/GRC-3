    54→/** Get framework names string from a quote (uses join table, falls back to single framework) */
    55→function getQuoteFrameworkString(q: Quote): string {
    56→  if (q.quoteFrameworks && q.quoteFrameworks.length > 0) {
    57→    return q.quoteFrameworks
    58→      .sort((a, b) => a.sortOrder - b.sortOrder)
    59→      .map((qf) => qf.framework?.name || 'Unknown')
    60→      .join(' + ')
    61→  }
    62→  return q.framework?.name || ''
    63→}

   495→                          <div className="flex items-center gap-2">
   496→                            <p className="text-sm font-medium text-slate-900 truncate">
   497→                              {q.client?.companyName || 'Unknown client'}
   498→                            </p>
   499→                            <StatusBadge status={q.status} />
   500→                          </div>
   501→                          <p className="text-xs text-slate-500 mt-0.5">
   502→                            {getQuoteFrameworkString(q)} • {formatRelativeTime(q.createdAt)}
   503→                          </p>
   504→                        </div>
   505→                        <div className="text-right shrink-0">
   506→                          <p className="text-sm font-bold text-slate-900">{formatINR(q.totalInr)}</p>
   507→                          <p className="text-[11px] text-slate-500">v{q.version}</p>
   508→                        </div>
   509→                      </motion.button>
   510→                    ))
   511→                  )}
   512→                </div>
   513→              </div>
   514→            </div>
   515→
   516→            {/* Export CSV */}
   517→            <div className="flex justify-end">
   518→              <button
   519→                onClick={exportCsv}
   520→                disabled={recentQuotes.length === 0}
   521→                className="inline-flex items-center gap-2 rounded-lg border border-slate-200 px-4 py-2 text-sm font-medium text-slate-700 hover:bg-slate-50 disabled:opacity-50 transition-colors"
   522→              >
   523→                <Download className="h-4 w-4" /> Export CSV
   524→              </button>
   525→            </div>
   526→          </>
   527→        )}
   528→      </div>
   529→
   530→      {/* Add Client Dialog */}
   531→      {addClientOpen && <AddClientDialog onClose={() => setAddClientOpen(false)} />}
   532→    </>
   533→  )
   534→}

   536→// ─── Animated Counter ──────────────────────────────────────────
   537→function AnimatedCounter({ value, duration = 1200 }: { value: number; duration?: number }) {
   538→  const [display, setDisplay] = useState(0)
   539→  const prevValue = useRef(0)
   540→
   541→  useEffect(() => {
   542→    const start = prevValue.current
   543→    const end = value
   544→    const startTime = performance.now()
   545→
   546→    const animate = (currentTime: number) => {
   547→      const elapsed = currentTime - startTime
   548→      const progress = Math.min(elapsed / duration, 1)
   549→      // Ease out cubic
   550→      const eased = 1 - Math.pow(1 - progress, 3)
   551→      setDisplay(Math.round(start + (end - start) * eased))
   552→      if (progress < 1) requestAnimationFrame(animate)
   553→    }
   554→
   555→    requestAnimationFrame(animate)
   556→    prevValue.current = value
   557→  }, [value, duration])
   558→
   559→  return <>{formatNumberIN(display)}</>
   560→}

   562→// ─── Export CSV ────────────────────────────────────────────────
   563→function exportCsv() {
   564→  const rows = [
   565→    ['Client', 'Framework', 'Tier', 'Status', 'Subtotal (INR)', 'Discount (INR)', 'GST (INR)', 'Total (INR)', 'USD', 'Created'],
   566→  ]
   567→
   568→  // We'll fetch fresh data for export
   569→  fetch('/api/quotes?status=all')
   570→    .then((r) => r.json())
   571→    .then((res) => {
   572→      const quotes = res.data || []
   573→      for (const q of quotes) {
   574→        rows.push([
   575→          q.client?.companyName || '',
   576→          getQuoteFrameworkString(q) || '',
   577→          q.tier?.name || '',
   578→          q.status,
   579→          String(q.subtotalInr),
   580→          String(q.discountInr),
   581→          String(q.gstAmountInr),
   582→          String(q.totalInr),
   583→          String(q.totalUsd),
   584→          new Date(q.createdAt).toLocaleDateString('en-IN'),
   585→        ])
   586→      }
   587→      const csv = rows.map((r) => r.map((c) => `"${c}"`).join(',')).join('\n')
   588→      const blob = new Blob([csv], { type: 'text/csv' })
   589→      const url = URL.createObjectURL(blob)
   590→      const a = document.createElement('a')
   591→      a.href = url
   592→      a.download = `isecurify-quotes-${new Date().toISOString().split('T')[0]}.csv`
   593→      document.body.appendChild(a)
   594→      a.click()
   595→      document.body.removeChild(a)
   596→      URL.revokeObjectURL(url)
   597→    })
   598→    .catch(() => {
   599→      // fallback: ignore
   600→    })
   601→}
