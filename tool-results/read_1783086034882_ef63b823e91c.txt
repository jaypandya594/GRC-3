    59→async function downloadQuotePdf(id: string): Promise<void> {
    60→  try {
    61→    const win = window.open(`/api/quote-pdf/${id}`, '_blank')
    62→    if (!win) {
    63→      const res = await fetch(`/api/quote-pdf/${id}`)
    64→      if (!res.ok) throw new Error(`Server returned ${res.status}`)
    65→      const blob = await res.blob()
    66→      const blobUrl = URL.createObjectURL(blob)
    67→      const a = document.createElement('a')
    68→      a.style.display = 'none'
    69→      a.href = blobUrl
    70→      a.download = `iSecurify-Quote-${id.slice(-8).toUpperCase()}.pdf`
    71→      document.body.appendChild(a)
    72→      a.click()
    73→      setTimeout(() => {
    74→        document.body.removeChild(a)
    75→        URL.revokeObjectURL(blobUrl)
    76→      }, 5000)
    77→    }
    78→  } catch (err) {
    79→    console.error('[PDF Download] Error:', err)
    80→    throw err
    81→  }
    82→}

    86→/** Get all framework names from the QuoteFramework join table, falling back to line items, then single framework */
    87→function getFrameworkDisplayNames(quote: Quote): string[] {
    88→  // 1. Prefer the join table
    89→  if (quote.quoteFrameworks && quote.quoteFrameworks.length > 0) {
    90→    return quote.quoteFrameworks
    91→      .sort((a, b) => a.sortOrder - b.sortOrder)
    92→      .map((qf) => qf.framework?.name || 'Unknown')
    93→  }
    94→  // 2. Fall back to line items
    95→  const consultingLines = (quote.lineItems || []).filter((l) => l.lineType === 'consulting_fee')
    96→  if (consultingLines.length > 0) {
    97→    return consultingLines.map((l) => l.description.replace(/ — Consulting Fee.*/, '').trim())
    98→  }
    99→  // 3. Last resort: single framework field
   100→  if (quote.framework?.name) return [quote.framework.name]
   101→  return []
   102→}
   103→
   104→function getFrameworkDisplayString(quote: Quote): string {
   105→  return getFrameworkDisplayNames(quote).join(' + ') || '—'
   106→}
   107→
   108→function FrameworkBadges({ quote }: { quote: Quote }) {
   109→  const names = getFrameworkDisplayNames(quote)
   110→  if (names.length === 0) return <span className="text-slate-400">—</span>
   111→  if (names.length <= 2) {
   112→    return <span className="text-slate-600">{names.join(' + ')}</span>
   113→  }
   114→  // Show first 2 + count for long lists
   115→  return (
   116→    <span className="text-slate-600">
   117→      {names.slice(0, 2).join(' + ')} <span className="text-slate-400">+{names.length - 2} more</span>
   118→    </span>
   119→  )
   120→}

   512→

   530→        <div className="lg:col-span-2 space-y-6">

   532→          <div className="rounded-xl border border-slate-200 bg-white p-5">
   533→            <div className="flex flex-wrap items-start justify-between gap-3 mb-4">

   556→          {/* Line items */}

   576→            </div>

   578→            {/* Totals */}
   579→            <div className="mt-4 pt-4 border-t border-slate-200 space-y-1.5">
   580→              <div className="flex justify-between text-sm">
   581→                <span className="text-slate-600">Subtotal</span>
   582→                <span className="font-medium text-slate-900">{formatINR(quote.subtotalInr)}</span>
   583→              </div>
   584→              {quote.discountInr > 0 && (
   585→                <div className="flex justify-between text-sm">
   586→                  <span className="text-rose-600">Discount ({quote.discountPct}%) {quote.discountReason ? `(${quote.discountReason})` : ''}</span>

   589→              )}
   590→              <div className="flex justify-between text-sm">
   591→                <span className="text-slate-600">GST @ {quote.gstRateSnapshot}%</span>
   592→                <span className="font-medium text-slate-900">{formatINR(quote.gstAmountInr)}</span>

   594→              <div className="flex justify-between items-baseline pt-2 border-t border-slate-200">

   600→              </div>
   601→            </div>
   602→
   603→            {/* Internal time (advisory) */}

   606→                <p className="text-xs font-semibold text-slate-600 mb-1">Advisory (not billed)</p>
   607→                {internalLines.map((line) => (
   608→                  <div key={line.id} className="flex justify-between text-xs text-slate-500 italic">
   609→                    <span>{line.description}</span>
   610→                    <span className="tabular-nums">{formatINR(line.amountInr)}</span>

   613→              </div>

   615→          </SectionCard>
   616→
   617→          {/* Approval timeline */}
   618→          <SectionCard title="Approval Timeline" description="Audit trail of every state transition">
   619→            {quote.approvals && quote.approvals.length > 0 ? (

   622→                  <div key={a.id} className="flex gap-3">

   629→                    <div className="flex-1 pb-3">

   641→                    </div>
   642→                  </div>

   647→            )}
   648→          </SectionCard>
   649→        </div>
   650→
   651→        {/* Right: actions */}
   652→        <div className="space-y-6">
   653→          {/* Edit Quote button */}
   654→          {canEdit && (
   655→            <button
   656→              onClick={handleEdit}
   657→              className="w-full inline-flex items-center justify-center gap-2 rounded-lg border border-brand-200 bg-brand-50 px-4 py-2.5 text-sm font-semibold text-brand-700 hover:bg-brand-100 transition-colors"
   658→            >
   659→              <Pencil className="h-4 w-4" />
   660→              Edit Quote
   661→            </button>
   662→          )}
   663→
   664→          {/* Actions */}
   665→          <SectionCard title="Actions" description={`Current status: ${quote.status.replace(/_/g, ' ')}`}>
   666→            <div className="space-y-2">
   667→              {availableActions.length === 0 ? (
   668→                <p className="text-sm text-slate-400 py-4 text-center">
   669→                  No actions available in current status
   670→                </p>
   671→              ) : (
   672→                availableActions.map((act) => (
   673→                  <button
   674→                    key={act.action}
   675→                    onClick={() => {
   676→                      if (act.action === 'override' && overrideDiscount === null) {
   677→                        setOverrideDiscount(quote.discountPct || 0)

   682→                        action: act.action,
   683→                        comment: comment || undefined,

   695→                    {act.icon} {act.label}

   699→

   701→              {overrideDiscount !== null && (

   703→                  <label className="text-xs font-medium text-amber-800">Override discount (%)</label>
   704→                  <input
   705→                    type="number"
   706→                    value={overrideDiscount}
   707→                    onChange={(e) => setOverrideDiscount(parseFloat(e.target.value) || 0)}
   708→                    max="100"
   709→                    step="0.5"
   710→                    className="w-full rounded border border-amber-300 px-2 py-1.5 text-sm"
   711→                  />
   712→                  <div className="flex gap-2">
   713→                    <button
   714→                      onClick={() => actionMutation.mutate({
   715→                        action: 'override',
   716→                        comment: comment || undefined,
   717→                        newDiscountPct: overrideDiscount,
   718→                      })}
   719→                      disabled={actionMutation.isPending}
   720→                      className="flex-1 rounded bg-amber-600 text-white px-3 py-1.5 text-xs font-semibold hover:bg-amber-700"
   721→                    >
   722→                      Confirm Override
   723→                    </button>
   724→                    <button
   725→                      onClick={() => setOverrideDiscount(null)}
   726→                      className="rounded border border-slate-200 px-3 py-1.5 text-xs"
   727→                    >
   728→                      Cancel
   729→                    </button>
   730→                  </div>
   731→                </div>
   732→              )}
   733→            </div>
   734→
   735→            {/* Comment */}
   736→            <div className="mt-3">
   737→              <label className="text-xs font-medium text-slate-600">Comment (optional)</label>
   738→              <textarea
   739→                value={comment}
   740→                onChange={(e) => setComment(e.target.value)}
   741→                rows={2}
   742→                placeholder="Add a note for the audit trail..."
   743→                className="mt-1 w-full rounded-lg border border-slate-200 px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-brand-600"
   744→              />
   745→            </div>
   746→          </SectionCard>
   747→
   748→          {/* Payment terms */}
   749→          <SectionCard title="Payment Terms">
   750→            <div className="space-y-2 text-sm">
   751→              <div className="flex justify-between">
   752→                <span className="text-slate-600">Advance (40%)</span>
   753→                <span className="font-medium text-slate-900">{formatINR(quote.totalInr * 0.4)}</span>
   754→              </div>
   755→              <div className="flex justify-between">
   756→                <span className="text-slate-600">Midway (40%)</span>
   757→                <span className="font-medium text-slate-900">{formatINR(quote.totalInr * 0.4)}</span>
   758→              </div>
   759→              <div className="flex justify-between">
   760→                <span className="text-slate-600">On Completion (20%)</span>
   761→                <span className="font-medium text-slate-900">{formatINR(quote.totalInr * 0.2)}</span>
   762→              </div>
   763→              <div className="pt-2 border-t border-slate-200 text-xs text-slate-500">
   764→                <Clock className="inline h-3 w-3 mr-1" />
   765→                Valid until {formatDate(quote.validUntil)}
   766→              </div>
   767→            </div>
   768→          </SectionCard>
   769→
   770→          {/* Download PDF */}
   771→          <button
   772→            onClick={handleDownloadPdf}
   773→            disabled={downloading}
   774→            className="w-full inline-flex items-center justify-center gap-2 rounded-lg bg-brand-700 px-4 py-2.5 text-sm font-semibold text-white hover:bg-brand-800 disabled:opacity-50"
   775→          >
   776→            {downloading ? <Loader2 className="h-4 w-4 animate-spin" /> : <Download className="h-4 w-4" />}
   777→            {downloading ? 'Generating...' : 'Download PDF'}
   778→          </button>
   779→
   780→          {/* Delete Quote — super_admin only per v2 spec */}
   781→          {role === 'super_admin' && (
   782→          <button
   783→            onClick={() => setShowDeleteConfirm(true)}
   784→            className="w-full inline-flex items-center justify-center gap-2 rounded-lg border border-rose-200 bg-rose-50 px-4 py-2.5 text-sm font-semibold text-rose-700 hover:bg-rose-100 transition-colors"

   786→            <Trash2 className="h-4 w-4" />
   787→            Delete Quote
   788→          </button>

   790→
   791→          {/* Delete Confirmation Dialog */}
   792→          <AlertDialog open={showDeleteConfirm} onOpenChange={setShowDeleteConfirm}>

   795→                <AlertDialogTitle>Delete this Quote?</AlertDialogTitle>
   796→                <AlertDialogDescription>
   797→                  This will permanently delete this quote and all its line items. This action cannot be undone.
   798→                </AlertDialogDescription>
   799→              </AlertDialogHeader>
   800→              <AlertDialogFooter>
   801→                <AlertDialogCancel disabled={deleteMutation.isPending}>Cancel</AlertDialogCancel>
   802→                <AlertDialogAction
   803→                  onClick={() => deleteMutation.mutate()}
   804→                  disabled={deleteMutation.isPending}
   805→                  className="bg-rose-600 hover:bg-rose-700 text-white"

   807→                  {deleteMutation.isPending && <Loader2 className="h-4 w-4 mr-1 animate-spin" />}

   810→              </AlertDialogFooter>

   813→

   815→          <SectionCard title="Upload Document" description="Attach supporting files">

   817→              <div className="relative">

   819→                  ref={fileInputRef}
   820→                  type="file"
   821→                  accept=".pdf,.doc,.docx,.xls,.xlsx"

   823→                  disabled={uploading}

   827→                  <div className="absolute inset-0 flex items-center justify-center bg-white/60 rounded-lg">

   829→                  </div>
   830→                )}
   831→              </div>
   832→              <p className="text-xs text-slate-400">Accepted: PDF, DOC, DOCX, XLS, XLSX</p>
   833→            </div>
   834→          </SectionCard>
   835→        </div>
   836→      </div>
   837→    </>
   838→  )
   839→}

   843→function buildQuoteSubtitle(quote: Quote, fwNames: string[]): string {
   844→  const fwDisplay = fwNames.join(' + ')
   845→  const fwShort = fwDisplay.length > 50 ? fwDisplay.slice(0, 47) + '...' : fwDisplay
   846→  return `${fwShort} • ${quote.tier?.name || '—'} • v${quote.version}`
   847→}
   848→
   849→function Meta({ label, value }: { label: string; value: string }) {
   850→  return (
   851→    <div>
   852→      <p className="text-xs text-slate-500">{label}</p>
   853→      <p className="text-sm font-medium text-slate-900 truncate">{value}</p>
   854→    </div>
   855→  )
   856→}
   857→
   858→function LineTypeIcon({ type }: { type: string }) {
   859→  const icons: Record<string, React.ReactNode> = {
   860→    consulting_fee: <FileText className="h-4 w-4 text-brand-600" />,
   861→    auditor_fee: <Check className="h-4 w-4 text-violet-600" />,
   862→    addon_service: <Plus className="h-4 w-4 text-amber-600" />,
   863→    grc_tool: <TrendingUp className="h-4 w-4 text-cyan-600" />,
   864→    dpo_vciso: <Clock className="h-4 w-4 text-rose-600" />,
   865→    retainer: <RotateCcw className="h-4 w-4 text-slate-600" />,
   866→    discount: <X className="h-4 w-4 text-rose-600" />,
   867→  }
   868→  return <span className="mt-0.5 shrink-0">{icons[type] || <FileText className="h-4 w-4 text-slate-400" />}</span>
   869→}
