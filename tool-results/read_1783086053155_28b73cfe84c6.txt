   350→   350→function SummaryPanel({ selection, pricing, calc }: {
   351→   351→  selection: QuoteBuilderSelection
   352→   352→  pricing: PricingData
   353→   353→  calc: ReturnType<typeof calculateQuote> | null
   354→   354→}) {
   355→   355→  if (!calc) return null
   356→   356→  const selectedFrameworks = pricing.frameworks.filter((f) => selection.selectedFrameworkIds.includes(f.id))
   357→   357→  const tier = pricing.tiers.find((t) => t.id === selection.tierId)
   358→   358→
   359→   359→  return (
   360→   360→    <div className="space-y-3">
   361→   361→      {calc.lines.filter((l) => l.lineType !== 'internal_time').length === 0 && (
   362→   362→        <p className="text-xs text-slate-400 py-4 text-center">
   363→   363→          Select framework(s) and a tier to see pricing
   364→   364→        </p>
   365→   365→      )}
   366→   366→      {selectedFrameworks.length > 0 && (
   367→   367→        <p className="text-xs font-medium text-brand-700">
   368→   368→          {selectedFrameworks.length} framework{selectedFrameworks.length > 1 ? 's' : ''} selected
   369→   369→        </p>
   370→   370→      )}
   371→   371→
   372→   372→      {calc.lines.map((line, i) => {
   373→   373→        const isInternal = line.lineType === 'internal_time'
   374→   374→        const isDiscount = line.lineType === 'discount'
   375→   375→        return (
   376→   376→          <div
   377→   377→            key={i}
   378→   378→            className={cn(
   379→   379→              'flex items-start justify-between gap-2 text-sm',
   380→   380→              isInternal && 'opacity-60',
   381→   381→            )}
   382→   382→          >
   383→   383→            <span className={cn('text-slate-600 flex-1', isInternal && 'italic text-xs', isDiscount && 'text-rose-600')}>
   384→   384→              {line.description}
   385→   385→            </span>
   386→   386→            <span className={cn(
   387→   387→              'font-medium tabular-nums shrink-0',
   388→   388→              isInternal ? 'text-slate-400 text-xs' : isDiscount ? 'text-rose-600' : 'text-slate-900',
   389→   389→            )}>
   390→   390→              {line.amountInr < 0 ? '-' : ''}{formatINR(Math.abs(line.amountInr))}
   391→   391→            </span>
   392→   392→          </div>
   393→   393→        )
   394→   394→      })}
   395→   395→
   396→   396→      <div className="border-t border-slate-200 pt-3 space-y-1.5">
   397→   397→        <div className="flex justify-between text-sm">
   398→   398→          <span className="text-slate-600">Subtotal</span>
   399→   399→          <span className="font-medium text-slate-900">{formatINR(calc.subtotalInr)}</span>
   400→   400→        </div>
   401→   401→        {calc.discountInr > 0 && (
   402→   402→          <div className="flex justify-between text-sm">
   403→   403→            <span className="text-rose-600">Discount ({calc.discountPct}%)</span>
   404→   404→            <span className="font-medium text-rose-600">-{formatINR(calc.discountInr)}</span>
   405→   405→          </div>
   406→   406→        )}
   407→   407→        <div className="flex justify-between text-sm">
   408→   408→          <span className="text-slate-600">GST @ {calc.gstRate}%</span>
   409→   409→          <span className="font-medium text-slate-900">{formatINR(calc.gstAmountInr)}</span>
   410→   410→        </div>
   411→   411→        <div className="flex justify-between items-baseline pt-2 border-t border-slate-200">
   412→   412→          <span className="text-sm font-bold text-slate-900">Grand Total</span>
   413→   413→          <div className="text-right">
   414→   414→            <p className="text-lg font-bold text-brand-700">{formatINR(calc.totalInr)}</p>
   415→   415→            <p className="text-xs text-slate-500">{formatUSD(calc.totalUsd)} @ ₹{calc.usdInrRate}</p>
   416→   416→          </div>
   417→   417→        </div>
   418→   418→        {calc.internalTimeCostInr > 0 && (
   419→   419→          <p className="text-xs text-slate-400 italic pt-1">
   420→   420→            Advisory (not billed): {formatINR(calc.internalTimeCostInr)}
   421→   421→          </p>
   422→   422→        )}
   423→   423→      </div>
   424→   424→    </div>
   425→   425→  )
   426→   426→}

   429→   429→function FxRateCard({ currentRate }: { currentRate: number }) {
   430→   430→  const queryClient = useQueryClient()
   431→   431→  const [editing, setEditing] = useState(false)
   432→   432→  const [draft, setDraft] = useState(String(currentRate))

   451→   451→  }

   471→   471→              onClick={() => { setDraft(String(currentRate)); setEditing(true) }}
   472→   472→              className="inline-flex items-center gap-1.5 rounded-lg border border-brand-200 bg-brand-50 px-3 py-1.5 text-xs font-medium text-brand-700 hover:bg-brand-100 transition-colors"
   473→   473→            >
   474→   474→              <Pencil className="h-3 w-3" /> Update Rate
   475→   475→            </button>
   476→   476→          </div>
   477→   477→        ) : (
   478→   478→          <div className="flex items-center gap-2">
   479→   479→            <input

   481→   481→              step={0.01}
   482→   482→              min={0.01}
   483→   483→              value={draft}
   484→   484→              onChange={(e) => setDraft(e.target.value)}
   485→   485→              onKeyDown={(e) => { if (e.key === 'Enter') handleSave(); if (e.key === 'Escape') handleCancel() }}
   486→   486→              className="w-28 rounded-lg border border-slate-200 px-2.5 py-1.5 text-sm tabular-nums focus:outline-none focus:ring-2 focus:ring-brand-600"
