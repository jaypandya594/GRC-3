    80→  const calc = pricing

   251→// ─── Summary Panel ─────────────────────────────────────────────
   252→function SummaryPanel({ selection, pricing, calc }: {
   253→  selection: QuoteBuilderSelection
   254→  pricing: PricingData
   255→  calc: ReturnType<typeof calculateQuote> | null
   256→}) {
   257→  if (!calc) return null
   258→  const framework = pricing.frameworks.find((f) => f.id === selection.frameworkId)
   259→  const tier = pricing.tiers.find((t) => t.id === selection.tierId)
   260→
   261→  return (
   262→    <div className="space-y-3">
   263→      {calc.lines.filter((l) => l.lineType !== 'internal_time').length === 0 && (
   264→        <p className="text-xs text-slate-400 py-4 text-center">
   265→          Select a framework and tier to see pricing
   266→        </p>
   267→      )}
   268→
   269→      {calc.lines.map((line, i) => {
   270→        const isInternal = line.lineType === 'internal_time'
   271→        const isDiscount = line.lineType === 'discount'
   272→        return (
   273→          <div
   274→            key={i}
   275→            className={cn(
   276→              'flex items-start justify-between gap-2 text-sm',
   277→              isInternal && 'opacity-60',
   278→            )}
   279→          >
   280→            <span className={cn('text-slate-600 flex-1', isInternal && 'italic text-xs', isDiscount && 'text-rose-600')}>
   281→              {line.description}
   282→            </span>
   283→            <span className={cn(
   284→              'font-medium tabular-nums shrink-0',
   285→              isInternal ? 'text-slate-400 text-xs' : isDiscount ? 'text-rose-600' : 'text-slate-900',
   286→            )}>
   287→              {line.amountInr < 0 ? '-' : ''}{formatINR(Math.abs(line.amountInr))}
   288→            </span>
   289→          </div>
   290→        )
   291→      })}
   292→
   293→      <div className="border-t border-slate-200 pt-3 space-y-1.5">
   294→        <div className="flex justify-between text-sm">
   295→          <span className="text-slate-600">Subtotal</span>
   296→          <span className="font-medium text-slate-900">{formatINR(calc.subtotalInr)}</span>
   297→        </div>
   298→        {calc.discountInr > 0 && (
   299→          <div className="flex justify-between text-sm">
   300→            <span className="text-rose-600">Discount ({calc.discountPct}%)</span>
   301→            <span className="font-medium text-rose-600">-{formatINR(calc.discountInr)}</span>
   302→          </div>
   303→        )}
   304→        <div className="flex justify-between text-sm">
   305→          <span className="text-slate-600">GST @ {calc.gstRate}%</span>
   306→          <span className="font-medium text-slate-900">{formatINR(calc.gstAmountInr)}</span>
   307→        </div>
   308→        <div className="flex justify-between items-baseline pt-2 border-t border-slate-200">
   309→          <span className="text-sm font-bold text-slate-900">Grand Total</span>
   310→          <div className="text-right">
   311→            <p className="text-lg font-bold text-brand-700">{formatINR(calc.totalInr)}</p>
   312→            <p className="text-xs text-slate-500">{formatUSD(calc.totalUsd)} @ ₹{calc.usdInrRate}</p>
   313→          </div>
   314→        </div>
   315→        {calc.internalTimeCostInr > 0 && (
   316→          <p className="text-xs text-slate-400 italic pt-1">
   317→            Advisory (not billed): {formatINR(calc.internalTimeCostInr)}
   318→          </p>
   319→        )}
   320→      </div>
   321→    </div>
   322→  )
   323→}

   326→function FxRateCard({ currentRate }: { currentRate: number }) {
   327→  const queryClient = useQueryClient()
   328→  const [editing, setEditing] = useState(false)
   329→  const [draft, setDraft] = useState(String(currentRate))
   330→
   331→  const updateFx = useMutation({
   332→    mutationFn: (rate: number) => api.put('/admin/fx-rate', { rate }),
   333→    onSuccess: () => {
   334→      queryClient.invalidateQueries({ queryKey: ['pricing'] })
   335→      toast.success('FX rate updated')
   336→      setEditing(false)
   337→    },
   338→    onError: (e: Error) => toast.error(e.message),
   339→  })
   340→
   341→  const handleSave = () => {
   342→    const val = parseFloat(draft)
   343→    if (isNaN(val) || val <= 0) {
   344→      toast.error('Enter a valid positive rate')
   345→      return
   346→    }
   347→    updateFx.mutate(val)
   348→  }
   349→
   350→  const handleCancel = () => {
   351→    setDraft(String(currentRate))
   352→    setEditing(false)
   353→  }
   354→
   355→  return (
   356→    <div className="rounded-xl border border-slate-200 bg-white p-5">
   357→      <div className="flex items-start justify-between gap-4">
   358→        <div className="space-y-1">
   359→          <h3 className="text-sm font-semibold text-slate-900">USD/INR Exchange Rate</h3>
   360→          {!editing && (
   361→            <p className="text-xs text-slate-500">This rate will be used to calculate USD equivalents</p>
   362→          )}
   363→        </div>
   364→        {!editing ? (
   365→          <div className="flex items-center gap-3">
   366→            <span className="text-2xl font-bold text-brand-700 tabular-nums">{currentRate.toFixed(2)}</span>
   367→            <button
   368→              onClick={() => { setDraft(String(currentRate)); setEditing(true) }}
   369→              className="inline-flex items-center gap-1.5 rounded-lg border border-brand-200 bg-brand-50 px-3 py-1.5 text-xs font-medium text-brand-700 hover:bg-brand-100 transition-colors"
   370→            >
   371→              <Pencil className="h-3 w-3" /> Update Rate
   372→            </button>
   373→          </div>
   374→        ) : (
   375→          <div className="flex items-center gap-2">
   376→            <input
   377→              type="number"
   378→              step={0.01}
   379→              min={0.01}
   380→              value={draft}
   381→              onChange={(e) => setDraft(e.target.value)}
   382→              onKeyDown={(e) => { if (e.key === 'Enter') handleSave(); if (e.key === 'Escape') handleCancel() }}
   383→              className="w-28 rounded-lg border border-slate-200 px-2.5 py-1.5 text-sm tabular-nums focus:outline-none focus:ring-2 focus:ring-brand-600"
   384→              autoFocus
   385→            />
   386→            <button
   387→              onClick={handleSave}
   388→              disabled={updateFx.isPending}
   389→              className="inline-flex items-center gap-1 rounded-lg bg-brand-700 px-3 py-1.5 text-xs font-semibold text-white hover:bg-brand-800 disabled:opacity-50 transition-colors"
   390→            >
   391→              {updateFx.isPending ? <Loader2 className="h-3 w-3 animate-spin" /> : <Check className="h-3 w-3" />} Save
   392→            </button>
   393→            <button
   394→              onClick={handleCancel}
   395→              disabled={updateFx.isPending}
   396→              className="inline-flex items-center gap-1 rounded-lg border border-slate-200 px-3 py-1.5 text-xs font-medium text-slate-600 hover:bg-slate-50 disabled:opacity-50 transition-colors"
   397→            >
   398→              Cancel
   399→            </button>
   400→          </div>
   401→        )}
   402→      </div>
   403→    </div>
   404→  )
   405→}

   407→function StepFramework({ selection, update, pricing, clients }: {
   408→  selection: QuoteBuilderSelection
   409→  update: (p: Partial<QuoteBuilderSelection>) => void
   410→  pricing: PricingData
   411→  clients: Client[]
   412→}) {
   413→  const grouped = pricing.frameworks.reduce((acc, fw) => {
   414→    (acc[fw.category] ||= []).push(fw)
   415→    return acc
   416→  }, {} as Record<string, Framework[]>)
   417→
   418→  return (
   419→    <div className="space-y-6">
   420→      {/* FX Rate */}
   421→      <FxRateCard currentRate={pricing.fxRate} />
   422→
   423→      {/* Client selection */}
   424→      <SectionCard title="1. Select Client" description="Choose the prospect this quote is for">
   425→        <select
   426→          value={selection.clientId || ''}
   427→          onChange={(e) => update({ clientId: e.target.value || null })}
   428→          className="w-full rounded-lg border border-slate-200 px-3 py-2.5 text-sm focus:outline-none focus:ring-2 focus:ring-brand-600"
   429→        >
   430→          <option value="">— Select a client —</option>
   431→          {clients.map((c) => (
   432→            <option key={c.id} value={c.id}>
   433→              {c.companyName} ({c.sector || 'N/A'} • {c.companySize})
   434→            </option>
   435→          ))}
   436→        </select>
   437→      </SectionCard>
   438→
   439→      {/* Framework selection */}
   440→      <SectionCard title="2. Select Framework" description="Only one framework per quote — grouped by category">
   441→        {Object.entries(grouped).map(([category, fws]) => {
   442→          const meta = FRAMEWORK_CATEGORY_META[category] || { label: category, color: 'slate' }
   443→          return (
   444→            <div key={category} className="mb-6 last:mb-0">
   445→              <h4 className="text-xs font-bold uppercase tracking-wide text-slate-500 mb-3">{meta.label}</h4>
   446→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">
   447→                {fws.map((fw) => (
   448→                  <PricingCard
   449→                    key={fw.id}
   450→                    selected={selection.frameworkId === fw.id}
   451→                    onClick={() => update({ frameworkId: fw.id })}
   452→                    title={fw.name}
   453→                    description={fw.description || undefined}
   454→                    badge={meta.label}
   455→                  />
   456→                ))}
   457→              </div>
   458→            </div>
   459→          )
   460→        })}
   461→      </SectionCard>
   462→    </div>
   463→  )
   464→}

   466→// ─── Step 1: Tier ──────────────────────────────────────────────
   467→function StepTier({ selection, update, pricing }: {
   468→  selection: QuoteBuilderSelection
   469→  update: (p: Partial<QuoteBuilderSelection>) => void
   470→  pricing: PricingData
   471→}) {
   472→  const framework = pricing.frameworks.find((f) => f.id === selection.frameworkId)
   473→  if (!framework) {
   474→    return (
   475→      <SectionCard title="Select Tier">
   476→        <EmptyState title="No framework selected" description="Go back to Step 1 and select a framework first." />
   477→      </SectionCard>
   478→    )
   479→  }
   480→
   481→  return (
   482→    <SectionCard
   483→      title="3. Select Tier"
   484→      description={`${framework.name} — pricing varies by company size`}
   485→    >
   486→      <div className="grid grid-cols-1 md:grid-cols-3 gap-4 mb-6">
   487→        {pricing.tiers.map((tier) => {
   488→          const price = pricing.frameworkPrices.find(
   489→            (p) => p.frameworkId === framework.id && p.tierId === tier.id,
   490→          )
   491→          return (
   492→            <PricingCard
   493→              key={tier.id}
   494→              selected={selection.tierId === tier.id}
   495→              onClick={() => update({ tierId: tier.id })}
   496→              title={tier.name}
   497→              subtitle={`Retainer: ${tier.retainerPct}%`}
   498→              fee={price?.projectFeeInr}
   499→              feeLabel="project fee"
   500→              description={price ? `+ ${formatINR(price.retainerFeeInr)} annual retainer` : undefined}
   501→            />
   502→          )
   503→        })}
   504→      </div>
   505→
   506→      {selection.tierId && (
   507→        <div className="rounded-lg border border-slate-200 p-4 bg-slate-50">
   508→          <label className="flex items-center gap-3 cursor-pointer">
   509→            <input
   510→              type="checkbox"
   511→              checked={selection.includeRetainer}
   512→              onChange={(e) => update({ includeRetainer: e.target.checked })}
   513→              className="h-4 w-4 rounded border-slate-300 text-brand-600 focus:ring-brand-600"
   514→            />
   515→            <div>
   516→              <p className="text-sm font-medium text-slate-900">Include Annual Retainer</p>
   517→              <p className="text-xs text-slate-500">
   518→                Adds a separate retainer line item ({pricing.tiers.find((t) => t.id === selection.tierId)?.retainerPct}% of project fee)
   519→              </p>
   520→            </div>
   521→          </label>
   522→        </div>
   523→      )}
   524→    </SectionCard>
   525→  )
   526→}

   528→// ─── Step 2: Auditor Fees ──────────────────────────────────────
   529→function StepAuditor({ selection, update, pricing }: {
   530→  selection: QuoteBuilderSelection
   531→  update: (p: Partial<QuoteBuilderSelection>) => void
   532→  pricing: PricingData
   533→}) {

   538→  }, {} as Record<string, AuditorFee[]>)

   540→  const toggle = (id: string) => {
   541→    const ids = selection.selectedAuditorFeeIds
   542→    update({
   543→      selectedAuditorFeeIds: ids.includes(id)

   545→        : [...ids, id],
   546→    })
   547→  }
   548→
   549→  const clearAll = () => update({ selectedAuditorFeeIds: [] })
   550→
   551→  return (
   552→    <SectionCard
   553→      title="4. Select Auditor Fees"

   557→          <button onClick={clearAll} className="text-xs font-medium text-rose-600 hover:text-rose-700">
   558→            Clear All
   559→          </button>
   560→        )
   561→      }
   562→    >
   563→      <div className="space-y-5">
   564→        {Object.entries(grouped).map(([standard, fees]) => (

   566→            <h4 className="text-xs font-bold uppercase tracking-wide text-slate-500 mb-2">{standard}</h4>
   567→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">
   568→              {fees.map((fee) => (
   569→                <PricingCard
   570→                  key={fee.id}
   571→                  selected={selection.selectedAuditorFeeIds.includes(fee.id)}
   572→                  onClick={() => toggle(fee.id)}
   573→                  title={fee.accreditationBody}
   574→                  subtitle={fee.notes || undefined}
   575→                  fee={fee.feeInr}
   576→                />
   577→              ))}
   578→            </div>
   579→          </div>
   580→        ))}
   581→      </div>
   582→    </SectionCard>
